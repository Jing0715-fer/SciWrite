   103→  const [userDataOpen, setUserDataOpen] = React.useState(false);
   104→  const [llmConfigOpen, setLlmConfigOpen] = React.useState(false);

   127→  }, []);
   128→
   129→  // auto-select first project
   130→  React.useEffect(() => {
   131→    if (!activeProjectId && projectsQ.data?.projects.length) {

   135→

   141→  const references = React.useMemo(() => {
   142→    const map = new Map<string, any>();
   143→    for (const r of project?.references ?? []) {
   144→      const key = `${r.type}:${r.externalId || r.title}`;
   145→      if (!map.has(key)) map.set(key, r);
   146→    }
   147→    for (const p of paragraphs) {
   148→      for (const r of p.references || []) {
   149→        const key = `${r.type}:${r.externalId || r.title}`;
   150→        if (!map.has(key)) map.set(key, r);
   151→      }
   152→    }

   154→  }, [paragraphs, project?.references]);
   155→
   156→  // Derived progress stats
   157→  const progressStats = React.useMemo(() => {
   158→    const totalWords = paragraphs.reduce(
   159→      (sum, p) => sum + (p.wordCount || 0),
   160→      0

   163→      const matches = p.content?.match(
   164→        /\[(\d{1,3}(?:[,\-–\s]\d{1,3})*|[A-Z]{2,12}:\s?[^\]\n]{1,60})\]/g

   166→      return sum + (matches?.length || 0);

   172→      paragraphs.length > 0
   173→        ? Math.round((paragraphsCited / paragraphs.length) * 100)

   176→      (s, p) => s + (p.annotations?.filter((a: any) => !a.resolved).length || 0),

   180→      (s, p) => s + (p.annotations?.filter((a: any) => a.resolved).length || 0),

   185→      totalParagraphs: paragraphs.length,

   189→      resolvedAnnotations: resolved,

   209→      if (isTyping || meta || e.altKey) return;
   210→      if (!activeProjectId) return;

   213→        e.preventDefault();

   215→        setUnifiedWriteOpen(true);
   216→      } else if (k === "g") {
   217→        e.preventDefault();
   218→        setUnifiedWriteTab("gather");
   219→        setUnifiedWriteOpen(true);
   220→      } else if (k === "i") {
   221→        e.preventDefault();
   222→        setInsightsOpen(true);
   223→      } else if (k === "o") {
   224→        e.preventDefault();
   225→        setUnifiedWriteTab("outline");
   226→        setUnifiedWriteOpen(true);
   227→      } else if (k === "c" && paragraphs.length >= 2) {
   228→        e.preventDefault();
   229→        setUnifiedWriteTab("compose");
   230→        setUnifiedWriteOpen(true);
   231→      } else if (k === "f") {
   232→        e.preventDefault();
   233→        setUnifiedWriteTab("full");
   234→        setUnifiedWriteOpen(true);
   235→      } else if (k === "d") {
   236→        e.preventDefault();

   238→        document.documentElement.classList.toggle("dark", !isDark);
   239→        try {
   240→          localStorage.setItem("theme", isDark ? "light" : "dark");
   241→        } catch {}
   242→      }
   243→    };
   244→    window.addEventListener("keydown", handler);
   245→    return () => window.removeEventListener("keydown", handler);

   275→        ) : isMobile ? (
   276→          <div className="flex flex-col h-full rounded-xl border border-border/60 bg-card overflow-hidden shadow-md">

   278→            <div className="flex-1 min-h-0 overflow-hidden">
   279→              {mobilePanel === "projects" && (
   280→                <ProjectsSidebar
   281→                  projects={projects}

   283→                  onSelect={(id) => {

   288→                  onOpenArticle={(a) => setViewArticle(a as Article)}

   290→              )}
   291→              {mobilePanel === "workspace" && (

   299→                  onOpenCompose={() => { setUnifiedWriteTab("compose"); setUnifiedWriteOpen(true); }}
   300→                  onOpenGather={() => { setUnifiedWriteTab("gather"); setUnifiedWriteOpen(true); }}

   302→                  progressStats={progressStats}

   304→                  onWordGoalChange={setWordGoal}

   345→                      active

   350→                    <Icon className="h-4 w-4" />

   424→      <Footer onOpenPalette={() => setPaletteOpen(true)} />
   425→
   426→      {/* Modals — lazy-loaded heavy dialogs wrapped in Suspense */}
   427→      {activeProjectId && project && (

   429→        <UnifiedWritingDialog

   431→          onOpenChange={setUnifiedWriteOpen}

   436→          initialTab={unifiedWriteTab}
   437→        />
   438→        </React.Suspense>
   439→      )}
   440→      {viewArticle && (

   446→        />
   447→        </React.Suspense>
   448→      )}
   449→      {activeProjectId && (
   450→        <React.Suspense fallback={null}>
   451→        <InsightsDialog
   452→          open={insightsOpen}
   453→          onOpenChange={setInsightsOpen}
   454→          projectId={activeProjectId}
   455→        />
   456→        </React.Suspense>
   457→      )}
   458→      {activeProjectId && (
   459→        <React.Suspense fallback={null}>
   460→        <UserDataDialog
   461→          open={userDataOpen}
   462→          onOpenChange={setUserDataOpen}
   463→          projectId={activeProjectId}
   464→        />
   465→        </React.Suspense>
   466→      )}
   467→      <LLMConfigDialog
   468→        open={llmConfigOpen}
   469→        onOpenChange={setLlmConfigOpen}
   470→      />
   471→      <CommandPalette
   472→        open={paletteOpen}
   473→        onOpenChange={setPaletteOpen}
   474→        actions={[
   475→          {
   476→            id: "write",
   477→            label: t("cmd.writeParagraph"),

   479→            icon: <Sparkles className="h-3.5 w-3.5" />,
   480→            shortcut: "N",
   481→            onSelect: () => setWriteOpen(true),
   482→            group: t("cmd.groupWriting"),

   485→          {

   495→          {

   497→            label: t("cmd.composeArticle"),
   498→            hint: t("cmd.composeHint"),
   499→            icon: <Layers className="h-3.5 w-3.5" />,
   500→            shortcut: "C",
   501→            onSelect: () => setComposeOpen(true),
   502→            group: t("cmd.groupWriting"),

   504→          },
   505→          {
   506→            id: "insights",
   507→            label: t("cmd.projectInsights"),
   508→            hint: t("cmd.insightsHint"),
   509→            icon: <BarChart3 className="h-3.5 w-3.5" />,
   510→            shortcut: "I",
   511→            onSelect: () => setInsightsOpen(true),
   512→            group: t("cmd.groupProject"),
   513→            disabled: !activeProjectId,
   514→          },
   515→          {
   516→            id: "outline",
   517→            label: t("cmd.generateOutline"),
   518→            hint: t("cmd.outlineHint"),
   519→            icon: <ListTree className="h-3.5 w-3.5" />,

   521→            onSelect: () => setOutlineOpen(true),
   522→            group: t("cmd.groupWriting"),
   523→            disabled: !activeProjectId,

   956→function EmbeddedReviewWorkspace({ articleId, articleTitle, projectId }: { articleId?: string; articleTitle?: string; projectId: string }) {
   957→  const { t } = useI18n();
   958→  const qc = useQueryClient();
   959→  // Load saved review from DB on mount
   960→  const { data: savedReview } = useQuery({
   961→    queryKey: ["saved-review", articleId],
   962→    queryFn: () => articleId ? api.getSavedReview(articleId) : Promise.resolve({ notFound: true }),
   963→    enabled: !!articleId,
   964→    staleTime: Infinity,
   965→  });
   966→  const { data: reviewData } = useQuery({
   967→    queryKey: ["article-review", articleId],
   968→    queryFn: () => api.aiReview({ mode: "review", articleId: articleId! }),
   969→    enabled: false,
   970→    staleTime: Infinity,
   971→  });
   972→
   973→  const reviewMut = useMutation({
   974→    mutationFn: () => articleId ? api.aiReview({ mode: "review", articleId }) : Promise.reject(new Error("No article")),
   975→    onSuccess: (data) => {
   976→      qc.setQueryData(["article-review", articleId], data);
   977→      qc.invalidateQueries({ queryKey: ["saved-review", articleId] });
   978→      toast.success(t("toast.reviewCompleted"));
   979→    },
   980→    onError: (e: Error) => toast.error(e.message),
   981→  });
   982→
   983→  // Use freshly-run review if available, else saved from DB
   984→  const displayData = reviewData || (savedReview && !savedReview.notFound ? savedReview : null);
   985→
   986→  return (
   987→    <ScrollArea className="flex-1 min-h-0 scroll-academic">
   988→      <div className="px-5 py-4 max-w-2xl mx-auto">
   989→        {!displayData && !reviewMut.isPending && (
   990→          <div className="text-center py-12">
   991→            <div className="h-14 w-14 mx-auto rounded-2xl bg-primary/10 flex items-center justify-center mb-3">
   992→              <Gavel className="h-7 w-7 text-primary" />
   993→            </div>
   994→            <h3 className="text-sm font-semibold">{t("workspace.peerReviewTitle")}</h3>
   995→            <p className="text-xs text-muted-foreground mt-1 max-w-sm mx-auto mb-4">
   996→              {t("workspace.peerReviewDesc")}
