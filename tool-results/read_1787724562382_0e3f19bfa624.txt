    91→  const isMobile = useIsMobile();

    93→  // is replaced with a tab bar so each panel gets full width. Values:
    94→  // "projects" | "workspace" | "data".
    95→  const [mobilePanel, setMobilePanel] = React.useState<"projects" | "workspace" | "data">("workspace");
    96→  const [activeProjectId, setActiveProjectId] = React.useState<string | null>(null);
    97→  const [writeOpen, setWriteOpen] = React.useState(false);
    98→  const [composeOpen, setComposeOpen] = React.useState(false);
    99→  const [tipsOpen, setTipsOpen] = React.useState(false);
   100→  const [viewArticle, setViewArticle] = React.useState<Article | null>(null);
   101→  const [insightsOpen, setInsightsOpen] = React.useState(false);
   102→  const [paletteOpen, setPaletteOpen] = React.useState(false);
   103→  const [userDataOpen, setUserDataOpen] = React.useState(false);
   104→  const [llmConfigOpen, setLlmConfigOpen] = React.useState(false);
   105→  const [unifiedWriteOpen, setUnifiedWriteOpen] = React.useState(false);
   106→  const [unifiedWriteTab, setUnifiedWriteTab] = React.useState<"outline" | "gather" | "paragraph" | "compose" | "full">("outline");

   127→  }, []);
   128→
   129→  // auto-select first project
   130→  React.useEffect(() => {
   131→    if (!activeProjectId && projectsQ.data?.projects.length) {

   133→    }

   135→

   138→  const paragraphs = (project?.paragraphs ?? []) as any[];
   139→  const dataSources = project?.dataSources ?? [];
   140→  const articles = (project?.articles ?? []) as any[];
   141→  const references = React.useMemo(() => {
   142→    const map = new Map<string, any>();
   143→    for (const r of project?.references ?? []) {
   144→      const key = `${r.type}:${r.externalId || r.title}`;
   145→      if (!map.has(key)) map.set(key, r);
   146→    }
   147→    for (const p of paragraphs) {
   148→      for (const r of p.references || []) {
   149→        const key = `${r.type}:${r.externalId || r.title}`;
   150→        if (!map.has(key)) map.set(key, r);
   151→      }
   152→    }
   153→    return [...map.values()];
   154→  }, [paragraphs, project?.references]);
   155→
   156→  // Derived progress stats
   157→  const progressStats = React.useMemo(() => {
   158→    const totalWords = paragraphs.reduce(
   159→      (sum, p) => sum + (p.wordCount || 0),
   160→      0
   161→    );
   162→    const totalCitations = paragraphs.reduce((sum, p) => {
   163→      const matches = p.content?.match(
   164→        /\[(\d{1,3}(?:[,\-–\s]\d{1,3})*|[A-Z]{2,12}:\s?[^\]\n]{1,60})\]/g

   166→      return sum + (matches?.length || 0);

   171→    const citationCoverage =
   172→      paragraphs.length > 0
   173→        ? Math.round((paragraphsCited / paragraphs.length) * 100)
   174→        : 0;

   176→      (s, p) => s + (p.annotations?.filter((a: any) => !a.resolved).length || 0),
   177→      0
   178→    );
   179→    const resolved = paragraphs.reduce(
   180→      (s, p) => s + (p.annotations?.filter((a: any) => a.resolved).length || 0),
   181→      0
   182→    );

   184→      totalWords,
   185→      totalParagraphs: paragraphs.length,
   186→      totalCitations,
   187→      citationCoverage,
   188→      unresolvedAnnotations: unresolved,
   189→      resolvedAnnotations: resolved,

   203→      if (meta && e.key.toLowerCase() === "k") {
   204→        e.preventDefault();
   205→        setPaletteOpen((v) => !v);
   206→        return;
   207→      }
   208→      // Single-key shortcuts only when not typing and no modifier (except Shift)
   209→      if (isTyping || meta || e.altKey) return;
   210→      if (!activeProjectId) return;
   211→      const k = e.key.toLowerCase();
   212→      if (k === "n") {
   213→        e.preventDefault();
   214→        setUnifiedWriteTab("paragraph");
   215→        setUnifiedWriteOpen(true);
   216→      } else if (k === "g") {
   217→        e.preventDefault();
   218→        setUnifiedWriteTab("gather");
   219→        setUnifiedWriteOpen(true);
   220→      } else if (k === "i") {
   221→        e.preventDefault();
   222→        setInsightsOpen(true);
   223→      } else if (k === "o") {
   224→        e.preventDefault();
   225→        setUnifiedWriteTab("outline");
   226→        setUnifiedWriteOpen(true);
   227→      } else if (k === "c" && paragraphs.length >= 2) {
   228→        e.preventDefault();
   229→        setUnifiedWriteTab("compose");
   230→        setUnifiedWriteOpen(true);
   231→      } else if (k === "f") {
   232→        e.preventDefault();
   233→        setUnifiedWriteTab("full");
   234→        setUnifiedWriteOpen(true);
   235→      } else if (k === "d") {
   236→        e.preventDefault();
   237→        const isDark = document.documentElement.classList.contains("dark");
   238→        document.documentElement.classList.toggle("dark", !isDark);
   239→        try {
   240→          localStorage.setItem("theme", isDark ? "light" : "dark");
   241→        } catch {}
   242→      }
   243→    };
   244→    window.addEventListener("keydown", handler);
   245→    return () => window.removeEventListener("keydown", handler);
   246→  }, [activeProjectId, paragraphs.length]);

   274→          </div>
   275→        ) : isMobile ? (

   279→          <div className="flex flex-col h-full rounded-xl border border-border/60 bg-card overflow-hidden">
   280→            {/* Active panel content — full width */}
   281→            <div className="flex-1 min-h-0 overflow-hidden">
   282→              {mobilePanel === "projects" && (
   283→                <ProjectsSidebar
   284→                  projects={projects}
   285→                  activeId={activeProjectId}
   286→                  onSelect={(id) => {
   287→                    setActiveProjectId(id);
   288→                    setMobilePanel("workspace");
   289→                  }}
   290→                  articles={articles}
   291→                  onOpenArticle={(a) => setViewArticle(a as Article)}
   292→                />
   293→              )}
   294→              {mobilePanel === "workspace" && (
   295→                <WritingWorkspace
   296→                  project={project}
   297→                  paragraphs={paragraphs}
   298→                  articles={articles}
   299→                  references={references}
   300→                  activeProjectId={activeProjectId}
   301→                  onOpenWrite={() => { setUnifiedWriteTab("paragraph"); setUnifiedWriteOpen(true); }}
   302→                  onOpenCompose={() => { setUnifiedWriteTab("compose"); setUnifiedWriteOpen(true); }}
   303→                  onOpenGather={() => { setUnifiedWriteTab("gather"); setUnifiedWriteOpen(true); }}
   304→                  onOpenOutline={() => { setUnifiedWriteTab("outline"); setUnifiedWriteOpen(true); }}
   305→                  progressStats={progressStats}
   306→                  wordGoal={wordGoal}
   307→                  onWordGoalChange={setWordGoal}
   308→                  tipsOpen={tipsOpen}
   309→                  onTipsOpenChange={setTipsOpen}

   311→                  onOpenArticle={(a) => setViewArticle(a as Article)}

   341→                const Icon = tab.icon;

   343→                return (
   344→                  <button
   345→                    key={tab.id}
   346→                    onClick={() => setMobilePanel(tab.id)}

   353→                    <Icon className="h-4 w-4" />
   354→                    {tab.label}
   355→                  </button>

   357→              })}
   358→            </div>
   359→          </div>

   362→          direction="horizontal"

   371→              onSelect={setActiveProjectId}

   383→              articles={articles}

   420→              </div>
   421→            </div>
   422→          </ResizablePanel>
   423→        </ResizablePanelGroup>
   424→        )}
   425→      </main>
   426→
   427→      <Footer onOpenPalette={() => setPaletteOpen(true)} />
   428→
   429→      {/* Modals — lazy-loaded heavy dialogs wrapped in Suspense */}
   430→      {activeProjectId && project && (
   431→        <React.Suspense fallback={null}>
   432→        <UnifiedWritingDialog
   433→          open={unifiedWriteOpen}
   434→          onOpenChange={setUnifiedWriteOpen}
   435→          projectId={activeProjectId}
   436→          topic={project.topic}
   437→          field={project.field}

   439→          initialTab={unifiedWriteTab}
   440→        />
   441→        </React.Suspense>
   442→      )}
   443→      {viewArticle && (
   444→        <React.Suspense fallback={null}>
   445→        <ArticleViewerWithTabs
   446→          article={viewArticle}
   447→          projectId={activeProjectId!}
   448→          onClose={() => setViewArticle(null)}
   449→        />
   450→        </React.Suspense>
   451→      )}
   452→      {activeProjectId && (
   453→        <React.Suspense fallback={null}>
   454→        <InsightsDialog
   455→          open={insightsOpen}
   456→          onOpenChange={setInsightsOpen}
   457→          projectId={activeProjectId}
   458→        />
   459→        </React.Suspense>
   460→      )}
   461→      {activeProjectId && (
   462→        <React.Suspense fallback={null}>
   463→        <UserDataDialog
   464→          open={userDataOpen}
   465→          onOpenChange={setUserDataOpen}
   466→          projectId={activeProjectId}
   467→        />
   468→        </React.Suspense>
   469→      )}
   470→      <LLMConfigDialog
   471→        open={llmConfigOpen}
   472→        onOpenChange={setLlmConfigOpen}
   473→      />
   474→      <CommandPalette
   475→        open={paletteOpen}
   476→        onOpenChange={setPaletteOpen}
   477→        actions={[
   478→          {
   479→            id: "write",
   480→            label: t("cmd.writeParagraph"),
   481→            hint: t("cmd.writeHint"),
   482→            icon: <Sparkles className="h-3.5 w-3.5" />,

   484→            onSelect: () => setWriteOpen(true),
   485→            group: t("cmd.groupWriting"),
   486→            disabled: !activeProjectId,
   487→          },
   488→          {
   489→            id: "gather",
   490→            label: t("cmd.gatherSourcesAction"),

   494→            onSelect: () => setGatherOpen(true),
   495→            group: t("cmd.groupWriting"),
   496→            disabled: !activeProjectId,
   497→          },
   498→          {
   499→            id: "compose",
   500→            label: t("cmd.composeArticle"),
   501→            hint: t("cmd.composeHint"),
   502→            icon: <Layers className="h-3.5 w-3.5" />,
   503→            shortcut: "C",
   504→            onSelect: () => setComposeOpen(true),
   505→            group: t("cmd.groupWriting"),
   506→            disabled: paragraphs.length < 2,
   507→          },
   508→          {
   509→            id: "insights",
   510→            label: t("cmd.projectInsights"),
   511→            hint: t("cmd.insightsHint"),
   512→            icon: <BarChart3 className="h-3.5 w-3.5" />,
   513→            shortcut: "I",
   514→            onSelect: () => setInsightsOpen(true),
   515→            group: t("cmd.groupProject"),
   516→            disabled: !activeProjectId,
   517→          },
   518→          {
   519→            id: "outline",
   520→            label: t("cmd.generateOutline"),
   521→            hint: t("cmd.outlineHint"),

   954→function EmbeddedReviewWorkspace({ articleId, articleTitle, projectId }: { articleId?: string; articleTitle?: string; projectId: string }) {
   955→  const { t } = useI18n();
   956→  const qc = useQueryClient();
   957→  // Load saved review from DB on mount
   958→  const { data: savedReview } = useQuery({
   959→    queryKey: ["saved-review", articleId],
   960→    queryFn: () => articleId ? api.getSavedReview(articleId) : Promise.resolve({ notFound: true }),
   961→    enabled: !!articleId,
   962→    staleTime: Infinity,
   963→  });
   964→  const { data: reviewData } = useQuery({
   965→    queryKey: ["article-review", articleId],
   966→    queryFn: () => api.aiReview({ mode: "review", articleId: articleId! }),
   967→    enabled: false,
   968→    staleTime: Infinity,
   969→  });
   970→
   971→  const reviewMut = useMutation({
   972→    mutationFn: () => articleId ? api.aiReview({ mode: "review", articleId }) : Promise.reject(new Error("No article")),
   973→    onSuccess: (data) => {
   974→      qc.setQueryData(["article-review", articleId], data);
   975→      qc.invalidateQueries({ queryKey: ["saved-review", articleId] });
   976→      toast.success(t("toast.reviewCompleted"));
   977→    },
   978→    onError: (e: Error) => toast.error(e.message),
   979→  });
   980→
   981→  // Use freshly-run review if available, else saved from DB
   982→  const displayData = reviewData || (savedReview && !savedReview.notFound ? savedReview : null);
   983→
   984→  return (
   985→    <ScrollArea className="flex-1 min-h-0 scroll-academic">
   986→      <div className="px-5 py-4 max-w-2xl mx-auto">
   987→        {!displayData && !reviewMut.isPending && (
   988→          <div className="text-center py-12">
   989→            <div className="h-14 w-14 mx-auto rounded-2xl bg-primary/10 flex items-center justify-center mb-3">
   990→              <Gavel className="h-7 w-7 text-primary" />
   991→            </div>
   992→            <h3 className="text-sm font-semibold">{t("workspace.peerReviewTitle")}</h3>
   993→            <p className="text-xs text-muted-foreground mt-1 max-w-sm mx-auto mb-4">
   994→              {t("workspace.peerReviewDesc")}
   995→            </p>
   996→            <Button size="sm" className="gap-1.5 text-xs" onClick={() => reviewMut.mutate()} disabled={!articleId}>
   997→              <Gavel className="h-3.5 w-3.5" /> {t("workspace.runReviewBtn")}
   998→            </Button>
   999→          </div>
  1000→        )}
  1001→        {reviewMut.isPending && !displayData && (
  1002→          <div className="flex items-center justify-center py-12"><Loader2 className="h-6 w-6 animate-spin text-primary" /></div>
