   125→/**
   126→ * CitationHealthDashboard — a compact, project-level citation-accuracy
   127→ * summary that sits in the workspace header (between ProgressTracker and
   128→ * the workspace tabs).
   129→ *
   130→ * On mount (and on refresh) it calls /api/projects/[id]/citation-health,
   131→ * which runs the inline + post-compose audits across all paragraphs and
   132→ * articles. It surfaces:
   133→ *   - A 0–100 health score with an A–F grade badge
   134→ *   - Counts: total citations, references, blocking errors, warnings
   135→ *   - A progress bar showing % paragraphs that are clean
   136→ *   - A collapsible "worst offenders" list — the 5 paragraphs with the most
   137→ *     findings, each showing its top 3 issues (blocking/warning verdicts)
   138→ *
   139→ * Clicking a worst-offender row scrolls the workspace to that paragraph.
   140→ */
   141→export function CitationHealthDashboard({
   142→  projectId,
   143→  onJumpParagraph,
   144→}: {
   145→  projectId: string;
   146→  onJumpParagraph?: (paragraphId: string) => void;
   147→}) {
   148→  const [report, setReport] = React.useState<HealthReport | null>(null);
   149→  const [loading, setLoading] = React.useState(true);
   150→  const [error, setError] = React.useState<string | null>(null);
   151→  const [open, setOpen] = React.useState(false);
   152→  const [fixing, setFixing] = React.useState(false);
   153→  const [fixResult, setFixResult] = React.useState<{
   154→    totalFixed: number;
   155→    totalBefore: number;
   156→    paragraphsProcessed: number;
   157→  } | null>(null);
   158→  // Live progress for batch fix: { done, total, currentTitle }.
   159→  const [fixProgress, setFixProgress] = React.useState<{
   160→    done: number;
   161→    total: number;
   162→    currentTitle?: string;
   163→  } | null>(null);
   164→  // Per-paragraph fixing state — supports the "Fix this" button on each
   165→  // worst-offender row.
   166→  const [fixingParagraphId, setFixingParagraphId] = React.useState<string | null>(null);
   167→
   168→  const fetchHealth = React.useCallback(async () => {
   169→    setLoading(true);
   170→    setError(null);
   171→    try {
   172→      const res = await fetch(`/api/projects/${projectId}/citation-health`);
   173→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   174→      const data = (await res.json()) as HealthReport;
   175→      setReport(data);
   176→      // Auto-expand when there are blocking errors so the user sees them.
   177→      if (data.aggregate.totalBlocking > 0) setOpen(true);
   178→    } catch (err: any) {
   179→      setError(err?.message || "Failed to load citation health.");
   180→    } finally {
   181→      setLoading(false);
   182→    }
   183→  }, [projectId]);
   184→
   185→  // Fix a single paragraph by calling the paragraph-level auto-fix endpoint.
   186→  // Used by both the batch loop and the per-paragraph "Fix this" button.
   187→  const fixParagraph = React.useCallback(
   188→    async (paragraphId: string): Promise<{ fixed: number; before: number }> => {
   189→      // Fetch the paragraph's current blocking count (before fix).
   190→      const healthRes = await fetch(
   191→        `/api/paragraphs/${paragraphId}/validate-citations`
   192→      );
   193→      const beforeData = healthRes.ok ? await healthRes.json() : null;
   194→      const before = beforeData?.missingCount ?? 0;
   195→
   196→      const fixRes = await fetch(
   197→        `/api/paragraphs/${paragraphId}/auto-fix-citations`,
   198→        { method: "POST" }
   199→      );
   200→      if (!fixRes.ok) throw new Error(`auto-fix HTTP ${fixRes.status}`);
   201→      const fixData = await fixRes.json();
   202→
   203→      // Re-validate to get the after count.
   204→      const afterRes = await fetch(
   205→        `/api/paragraphs/${paragraphId}/validate-citations`
   206→      );
   207→      const afterData = afterRes.ok ? await afterRes.json() : null;
   208→      const after = afterData?.missingCount ?? before;
   209→      return { fixed: Math.max(0, before - after), before };
   210→    },
   211→    []
   212→  );
   213→
   214→  // Batch auto-fix: iterates all worst-offender paragraphs client-side so we
   215→  // can show live progress (done/total + current paragraph title). This is
   216→  // better than a single batch API call because the user sees incremental
   217→  // progress instead of an opaque spinner for potentially minutes.
   218→  const runBatchAutoFix = React.useCallback(async () => {
   219→    if (!report) return;
   220→    const offenders = report.worstOffenders.filter((p) => p.blockingCount > 0);
   221→    if (offenders.length === 0) return;
   222→    setFixing(true);
   223→    setFixResult(null);
   224→    setFixProgress({ done: 0, total: offenders.length });
   225→    let totalFixed = 0;
   226→    let totalBefore = 0;
   227→    let processed = 0;
   228→    try {
   229→      for (let i = 0; i < offenders.length; i++) {
   230→        const p = offenders[i];
   231→        setFixProgress({
   232→          done: i,
   233→          total: offenders.length,
   234→          currentTitle: p.title,
   235→        });
   236→        setFixingParagraphId(p.paragraphId);
   237→        try {
   238→          const result = await fixParagraph(p.paragraphId);
   239→          totalFixed += result.fixed;
   240→          totalBefore += result.before;
   241→          processed++;
   242→        } catch (err) {
   243→          // Continue with other paragraphs even if one fails.
   244→          console.error(`auto-fix failed for ${p.paragraphId}:`, err);
   245→        }
   246→      }
   247→      setFixResult({
   248→        totalFixed,
   249→        totalBefore,
   250→        paragraphsProcessed: processed,
   251→      });
   252→      // Re-fetch health to reflect the fixes.
   253→      await fetchHealth();
   254→    } catch (err: any) {
   255→      setError(err?.message || "Batch auto-fix failed.");
   256→    } finally {
   257→      setFixing(false);
   258→      setFixProgress(null);
   259→      setFixingParagraphId(null);
   260→    }
   261→  }, [report, fixParagraph, fetchHealth]);
   262→
   263→  // Per-paragraph "Fix this" — fixes a single worst-offender in place.
   264→  const fixSingleParagraph = React.useCallback(
   265→    async (paragraphId: string) => {
   266→      setFixingParagraphId(paragraphId);
   267→      try {
   268→        await fixParagraph(paragraphId);
   269→        await fetchHealth();
   270→      } catch (err: any) {
   271→        setError(err?.message || "Auto-fix failed for this paragraph.");
   272→      } finally {
   273→        setFixingParagraphId(null);
   274→      }
   275→    },
   276→    [fixParagraph, fetchHealth]
   277→  );
   278→
   279→  // Per-paragraph "Regenerate" — calls the regenerate endpoint which re-writes
   280→  // the paragraph content via LLM using the current reference list. This is
   281→  // a stronger fix than "Fix" (which only adds references): regenerate
   282→  // produces fresh body text with correct [n] citations. Used when the
   283→  // paragraph content itself is broken (wrong citations, out-of-range, etc.).
   284→  const [regeneratingParagraphId, setRegeneratingParagraphId] = React.useState<string | null>(null);
   285→  const regenerateParagraph = React.useCallback(
   286→    async (paragraphId: string) => {
   287→      setRegeneratingParagraphId(paragraphId);
   288→      try {
   289→        const res = await fetch(`/api/paragraphs/${paragraphId}/regenerate`, {
   290→          method: "POST",
   291→        });
   292→        if (!res.ok) throw new Error(`regenerate HTTP ${res.status}`);
   293→        await fetchHealth();
   294→      } catch (err: any) {
   295→        setError(err?.message || "Regenerate failed for this paragraph.");
   296→      } finally {
   297→        setRegeneratingParagraphId(null);
   298→      }
   299→    },
   300→    [fetchHealth]
   301→  );
   302→
   303→  // Batch regenerate — iterates ALL worst-offender paragraphs (those with
   304→  // blocking OR warning findings) client-side, calling regenerate on each.
   305→  // Mirrors runBatchAutoFix but uses the regenerate endpoint. Shows live
   306→  // progress (done/total + current paragraph title). The regenResult badge
   307→  // shows how many paragraphs were processed.
   308→  const [regenProgress, setRegenProgress] = React.useState<{
   309→    done: number;
   310→    total: number;
   311→    currentTitle?: string;
   312→  } | null>(null);
   313→  const [regenResult, setRegenResult] = React.useState<{
   314→    processed: number;
   315→    total: number;
   316→  } | null>(null);
   317→  // Confirmation dialog state for "Regenerate all" — rewriting all
   318→  // paragraphs is a destructive operation, so we double-confirm.
   319→  const [confirmRegen, setConfirmRegen] = React.useState(false);
   320→  const runBatchRegenerate = React.useCallback(async () => {
   321→    if (!report) return;
   322→    // Regenerate ALL paragraphs with findings (blocking OR warnings) —
   323→    // regenerate can fix both by rewriting with better citations.
   324→    const offenders = report.worstOffenders.filter(
   325→      (p) => p.blockingCount > 0 || p.warningCount > 0
   326→    );
   327→    if (offenders.length === 0) return;
   328→    setRegeneratingParagraphId("__batch__"); // sentinel: batch mode active
   329→    setRegenResult(null);
   330→    setRegenProgress({ done: 0, total: offenders.length });
   331→    let processed = 0;
   332→    try {
   333→      for (let i = 0; i < offenders.length; i++) {
   334→        const p = offenders[i];
   335→        setRegenProgress({
   336→          done: i,
   337→          total: offenders.length,
   338→          currentTitle: p.title,
   339→        });
   340→        setRegeneratingParagraphId(p.paragraphId);
   341→        try {
   342→          const res = await fetch(`/api/paragraphs/${p.paragraphId}/regenerate`, {
   343→            method: "POST",
   344→          });
   345→          if (!res.ok) throw new Error(`regenerate HTTP ${res.status}`);
   346→          processed++;
   347→        } catch (err) {
   348→          console.error(`regenerate failed for ${p.paragraphId}:`, err);
   349→        }
   350→      }
   351→      setRegenResult({ processed, total: offenders.length });
   352→      await fetchHealth();
   353→    } catch (err: any) {
   354→      setError(err?.message || "Batch regenerate failed.");
   355→    } finally {
   356→      setRegeneratingParagraphId(null);
   357→      setRegenProgress(null);
   358→    }
   359→  }, [report, fetchHealth]);
   360→
   361→  React.useEffect(() => {
   362→    fetchHealth();
   363→  }, [fetchHealth]);
   364→
   365→  if (loading) {
   366→    return (
   367→      <div className="flex items-center gap-2 px-3 py-2 text-[11px] text-muted-foreground acad-fade-in">
   368→        <div className="h-5 w-5 rounded-md bg-primary/15 text-primary flex items-center justify-center ring-academic">
   369→          <Loader2 className="h-3 w-3 animate-spin" />
   370→        </div>
   371→        <span className="font-serif-text tracking-tight">Analyzing citation health…</span>
   372→      </div>
   373→    );
   374→  }
   375→
   376→  if (error) {
   377→    return (
   378→      <div className="flex items-center gap-2 px-3 py-2 text-[11px] text-muted-foreground acad-fade-in">
   379→        <div className="h-5 w-5 rounded-md bg-amber-500/15 text-amber-600 dark:text-amber-400 flex items-center justify-center ring-academic">
   380→          <CircleAlert className="h-3 w-3" />
   381→        </div>
   382→        <span className="font-serif-text tracking-tight">Citation health unavailable</span>
   383→        <Button
   384→          variant="ghost"
   385→          size="sm"
   386→          className="h-5 px-1.5 text-[10px] hover:bg-amber-500/10 hover:text-amber-700 dark:hover:text-amber-400"
   387→          onClick={fetchHealth}
   388→        >
   389→          <RefreshCw className="h-3 w-3" /> Retry
   390→        </Button>
   391→      </div>
   392→    );
   393→  }
   394→
   395→  if (!report) return null;
   396→
   397→  const agg = report.aggregate;
   398→  const hasBlocking = agg.totalBlocking > 0;
   399→  const hasWarnings = agg.totalWarnings > 0;
   400→
   401→  const Icon = hasBlocking ? ShieldX : hasWarnings ? ShieldAlert : ShieldCheck;
   402→  const iconColor = hasBlocking
   403→    ? "text-red-600 dark:text-red-400"
   404→    : hasWarnings
   405→    ? "text-amber-600 dark:text-amber-400"
   406→    : "text-emerald-600 dark:text-emerald-400";
   407→
   408→  const cleanPct =
   409→    agg.totalParagraphs > 0
   410→      ? Math.round((agg.paragraphsClean / agg.totalParagraphs) * 100)
   411→      : 100;
   412→
   413→  return (
   414→    <div className="px-5 py-2.5 border-b border-border/40 glass-subtle flex items-center gap-3 flex-wrap acad-fade-in relative">
   415→      {/* Grade badge — polished scoreboard tile (surface-card + ring-academic for A) */}
   416→      <TooltipProvider delayDuration={200}>
   417→        <Tooltip>
   418→          <TooltipTrigger asChild>
   419→            <div
   420→              className={cn(
   421→                "flex items-center gap-1.5 px-2.5 py-1 rounded-lg border text-[11px] font-bold cursor-help transition-all hover:shadow-md hover:scale-[1.02] surface-card",
   422→                GRADE_COLORS[agg.grade],
   423→                agg.grade === "A" && "ring-academic",
   424→                hasBlocking && "animate-pulse"
   425→              )}
   426→            >
   427→              <Icon className="h-3.5 w-3.5" />
   428→              <span className="font-serif-text text-base leading-none tracking-tight">{agg.grade}</span>
   429→              <span className="font-sans text-[10px] font-medium opacity-80 tabular-nums">
   430→                {agg.healthScore}
   431→              </span>
   432→            </div>
   433→          </TooltipTrigger>
   434→          <TooltipContent side="bottom" className="text-[10px]">
   435→            <div className="space-y-0.5">
   436→              <p className="font-semibold">
   437→                Citation Health: {GRADE_LABELS[agg.grade]} ({agg.healthScore}/100)
   438→              </p>
   439→              <p className="text-muted-foreground">
   440→                Grade = 100 − (5×blocking + 1×warning).
   441→              </p>
   442→              <p className="text-muted-foreground">
   443→                A ≥90 · B ≥70 · C ≥50 · D ≥30 · F &lt;30
   444→              </p>
   445→            </div>
   446→          </TooltipContent>
   447→        </Tooltip>
   448→      </TooltipProvider>
   449→
   450→      {/* Quick stats — surface-card tiles with hover lift + tabular-nums */}
   451→      <div className="flex items-center gap-2 text-[10px]">
   452→        <span className="surface-card rounded-md px-2 py-0.5 flex items-center gap-1 transition-all hover:shadow-md hover:border-primary/30">
   453→          <FileText className="h-3 w-3 text-primary" />
   454→          <span className="font-serif-text font-semibold tabular-nums text-foreground">{agg.totalCitations}</span>
   455→          <span className="text-muted-foreground">citations</span>
   456→        </span>
   457→        <span className="surface-card rounded-md px-2 py-0.5 flex items-center gap-1 transition-all hover:shadow-md hover:border-primary/30">
   458→          <BookOpen className="h-3 w-3 text-teal-600 dark:text-teal-400" />
   459→          <span className="font-serif-text font-semibold tabular-nums text-foreground">{agg.totalReferences}</span>
   460→          <span className="text-muted-foreground">refs</span>
   461→        </span>
   462→        {agg.totalBlocking > 0 ? (
   463→          <span className="surface-card rounded-md px-2 py-0.5 flex items-center gap-1 transition-all hover:shadow-md hover:border-rose-400/50">
   464→            <CircleX className="h-3 w-3 text-red-600 dark:text-red-400" />
   465→            <span className="font-serif-text font-semibold tabular-nums text-foreground">{agg.totalBlocking}</span>
   466→            <span className="text-muted-foreground">blocking</span>
   467→          </span>
   468→        ) : (
   469→          <span className="surface-card rounded-md px-2 py-0.5 flex items-center gap-1 transition-all hover:shadow-md hover:border-emerald-400/50">
   470→            <span className="h-1.5 w-1.5 rounded-full bg-emerald-500" />
   471→            <span className="font-serif-text font-semibold text-emerald-700 dark:text-emerald-400">0 blocking</span>
   472→          </span>
   473→        )}
   474→        {agg.totalWarnings > 0 && (
   475→          <span className="surface-card rounded-md px-2 py-0.5 flex items-center gap-1 transition-all hover:shadow-md hover:border-amber-400/50">
   476→            <AlertTriangle className="h-3 w-3 text-amber-600 dark:text-amber-400" />
   477→            <span className="font-serif-text font-semibold tabular-nums text-foreground">{agg.totalWarnings}</span>
   478→            <span className="text-muted-foreground">warnings</span>
   479→          </span>
   480→        )}
   481→      </div>
   482→
   483→      {/* Clean progress — refined label typography */}
   484→      <div className="flex items-center gap-2 min-w-[160px] flex-1 max-w-[280px]">
   485→        <Progress
   486→          value={cleanPct}
   487→          className={cn(
   488→            "h-2 flex-1 transition-all",
   489→            cleanPct === 100
   490→              ? "[&>div]:bg-gradient-to-r [&>div]:from-emerald-400 [&>div]:to-emerald-600"
   491→              : cleanPct >= 60
   492→              ? "[&>div]:bg-gradient-to-r [&>div]:from-amber-400 [&>div]:to-amber-600"
   493→              : "[&>div]:bg-gradient-to-r [&>div]:from-red-400 [&>div]:to-red-600"
   494→          )}
   495→        />
   496→        <span className="text-[9px] text-muted-foreground font-serif-text tabular-nums shrink-0 font-medium tracking-tight">
   497→          {agg.paragraphsClean}/{agg.totalParagraphs} clean
   498→        </span>
   499→      </div>
   500→
   501→      {/* Expand button */}
   502→      {(report.worstOffenders.length > 0 || report.articles.length > 0) && (
   503→        <Collapsible open={open} onOpenChange={setOpen} className="ml-auto">
   504→          <CollapsibleTrigger asChild>
   505→            <Button variant="ghost" size="sm" className="h-6 px-2 text-[10px] gap-1 glass-subtle rounded-md hover:shadow-sm transition-all">
   506→              {open ? (
   507→                <ChevronDown className="h-3 w-3" />
   508→              ) : (
   509→                <ChevronRight className="h-3 w-3" />
   510→              )}
   511→              {report.worstOffenders.length > 0
   512→                ? `${report.worstOffenders.length} offender${
   513→                    report.worstOffenders.length === 1 ? "" : "s"
   514→                  }`
   515→                : "Details"}
   516→            </Button>
   517→          </CollapsibleTrigger>
   518→        </Collapsible>
   519→      )}
   520→
   521→      {/* Batch auto-fix button — runs the LLM + database query pipeline to
   522→          resolve missing/out-of-range citations across all offending
   523→          paragraphs. Only shown when there are blocking errors. During the
   524→          fix, shows live progress (done/total) instead of an opaque spinner. */}
   525→      {agg.totalBlocking > 0 && (
   526→        <div className="flex items-center gap-1.5">
   527→          <Button
   528→            variant="outline"
   529→            size="sm"
