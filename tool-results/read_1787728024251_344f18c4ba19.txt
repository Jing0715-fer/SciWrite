    14→async function jfetch<T>(url: string, init?: RequestInit): Promise<T> {
    15→  const res = await fetch(url, {
    16→    ...init,
    17→    headers: {
    18→      "Content-Type": "application/json",
    19→      ...(init?.headers || {}),
    20→    },
    21→  });
    22→  const text = await res.text();
    23→  let data: any = null;
    24→  try {
    25→    data = text ? JSON.parse(text) : null;
    26→  } catch {
    27→    data = text;
    28→  }
    29→  if (!res.ok) {
    30→    const msg =
    31→      (data && (data.error || data.message)) ||
    32→      `Request failed (${res.status})`;
    33→    throw new Error(typeof msg === "string" ? msg : JSON.stringify(msg));
    34→  }
    35→  return data as T;
    36→}

   459→  /* AI generate full article (streaming SSE — gather → plan → generate → compose) */
   460→  aiGenerateFullStream: (
   461→    input: {
   462→      projectId: string;
   463→      journalTemplate?: string;
   464→      language?: string;
   465→      targetWords?: number;
   466→      /** Advanced tuning — all optional, backend applies clamped defaults. */
   467→      maxDbQueries?: number;
   468→      maxWebSearchQueries?: number;
   469→      gatherJsonCharLimit?: number;
   470→      sectionRefTopN?: number;
   471→      sectionRefMinN?: number;
   472→      sectionDsTopN?: number;
   473→      sectionDsMinN?: number;
   474→      maxTokens?: number;
   475→      /** Custom instruction from a selected prompt template — appended to
   476→       *  the section-generation prompt to customize LLM behavior. */
   477→      promptInstruction?: string;
   478→    },
   479→    onEvent: (event: string, data: any) => void
   480→  ): Promise<any> => {
   481→    return new Promise(async (resolve, reject) => {
   482→      try {
   483→        const res = await fetch(`/api/ai/generate-full`, {
   484→          method: "POST",
   485→          headers: { "Content-Type": "application/json" },
   486→          body: JSON.stringify(input),
   487→        });
   488→
   489→        if (!res.ok) {
   490→          const t = await res.text();
   491→          throw new Error(t || `Generation failed (${res.status})`);
   492→        }
   493→
   494→        const reader = res.body?.getReader();
   495→        if (!reader) throw new Error("No response body");
   496→
   497→        const decoder = new TextDecoder();
   498→        let buffer = "";
   499→        let finalResult: any = null;
   500→
   501→        while (true) {
   502→          const { done, value } = await reader.read();
   503→          if (done) break;
   504→
   505→          buffer += decoder.decode(value, { stream: true });
   506→          const lines = buffer.split("\n\n");
   507→          buffer = lines.pop() || "";
   508→
   509→          for (const line of lines) {
   510→            if (!line.startsWith("data: ")) continue;
   511→            try {
   512→              const data = JSON.parse(line.slice(6));
   513→              onEvent(data.event, data);
   514→              if (data.event === "complete") {
   515→                finalResult = data;
   516→              }
   517→              if (data.event === "error") {
   518→                reject(new Error(data.error));
   519→                return;
   520→              }
   521→            } catch {}
   522→          }
   523→        }
   524→
   525→        resolve(finalResult);
   526→      } catch (e) {
   527→        reject(e);
   528→      }
   529→    });
   530→  },

   568→        let finalResult: any = null;
   569→
   570→        while (true) {
   571→          const { done, value } = await reader.read();
   572→          if (done) break;
   573→
   574→          buffer += decoder.decode(value, { stream: true });
   575→          const lines = buffer.split("\n\n");
   576→          buffer = lines.pop() || "";
   577→
   578→          for (const line of lines) {
   579→            if (!line.startsWith("data: ")) continue;
   580→            try {
   581→              const data = JSON.parse(line.slice(6));
   582→              onEvent(data.event, data);
   583→              if (data.event === "complete") {
   584→                finalResult = data;
   585→              }
   586→              if (data.event === "error") {
   587→                reject(new Error(data.error));
   588→                return;
   589→              }
   590→            } catch {}
   591→          }
   592→        }
   593→
   594→        resolve(finalResult);
   595→      } catch (e) {
   596→        reject(e);
   597→      }
   598→    });
   599→  },
