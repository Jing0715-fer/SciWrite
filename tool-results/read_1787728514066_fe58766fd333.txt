    39→/**
    40→ * generate-full-v2 — Evidence-grounded article generation pipeline.
    41→ *
    42→ * Multi-stage architecture (deepseek-harness-inspired: analyze → allocate →
    43→ * write, with a validation gate between every stage):
    44→ *
    45→ *   1. gather      — fresh multi-database + web retrieval (same as v1)
    46→ *   2. curate      — LLM selects the most relevant citable subset
    47→ *   3. plan        — LLM designs the section outline
    48→ *   4. analyze     — ★ NEW: extract a structured EVIDENCE BANK (claims
    49→ *                    pre-bound to their sources) from every curated ref
    50→ *   5. allocate    — ★ NEW: assign references + evidence to sections
    51→ *   6. generate    — ★ per-section writing with STRUCTURAL citation keys
    52→ *                    ({{Rn}}) — the LLM never writes numbers, so numbering
    53→ *                    cannot drift; a validation gate retries sections that
    54→ *                    leak raw numeric markers
    55→ *   7. verify      — ★ NEW: adversarial per-citation verification (does this
    56→ *                    specific reference support this specific claim?) with
    57→ *                    conservative removal of unsupported citations
    58→ *   8. compose     — global renumbering + reference sync + article save
    59→ *
    60→ * Accuracy contract (what v2 guarantees that v1 cannot):
    61→ *   - every [n] in the final article was produced by CODE from a {{Rn}} key
    62→ *     that the model copied from a specific reference entry
    63→ *   - every surviving citation survived an adversarial claim-level check
    64→ *     against the reference's own title/abstract
    65→ *   - paragraph reference lists are synced to global numbering (no drift)
    66→ */
    67→

   179→        const gatherSystem =
   180→          "You are a research data strategist. Design a COMPREHENSIVE multi-database search plan.";

   182→FIELD: ${project.field || "life sciences"}
   183→PURPOSE: Write a comprehensive review article (~${targetWords} words).

   187→

   190→  "queries": [
   191→    { "database": "pubmed", "query": "concrete search string", "rationale": "short reason" }

   197→          system: gatherSystem,
   198→          temperature: 0.4,
   199→          taskType: "gather",
   200→          maxTokens,

   226→
   227→        // Execute non-NCBI in parallel, NCBI sequentially (rate limit)

   229→        const otherQueries = dbQueries.filter((q: any) => q.database !== "pubmed" && q.database !== "ncbi");
   230→
   231→        const runWithRetry = async (database: string, query: string) => {
   232→          for (let attempt = 0; attempt < 3; attempt++) {
   233→            try {
   234→              return await queryDatabase(database as any, query);
   235→            } catch (err: any) {
   236→              if (err?.message?.includes("HTTP 400") || attempt >= 2) throw err;
   237→              await new Promise((r) => setTimeout(r, 1000 * Math.pow(2, attempt)));
   238→            }
   239→          }
   240→          throw new Error("unreachable");
   241→        };
   242→
   243→        const otherResults = await Promise.allSettled(
   244→          otherQueries.map(async (q: any) => {
   245→            const r = await runWithRetry(q.database, q.query);

   247→              step: "gather",
   248→              status: "progress",

   259→            const r = await runWithRetry(q.database, q.query);

   261→            send("step", {
   262→              step: "gather",
   263→              status: "progress",
   264→              message: `[${q.database}] "${q.query.slice(0, 50)}" → ${r.items?.length || 0} results`,
   265→              resultCount: r.items?.length || 0,
   266→            });
   267→          } catch (err: any) {
   268→            ncbiResults.push({ status: "rejected", reason: err });
   269→          }

   271→        }
   272→
   273→        const dbItems: any[] = [];
   274→        for (const r of [...otherResults, ...ncbiResults]) {

   291→            for (const item of searchResults) {
   292→              webItems.push({
   293→                source: "web",
   294→                externalId: item.url,
   295→                title: item.name || item.url,
   296→                authors: item.host_name || undefined,
   297→                year: item.date?.slice(0, 4) || undefined,
   298→                url: item.url,
   299→                abstract: item.snippet,
   300→                extra: { host: item.host_name, rank: item.rank },
   301→                queryUsed: "web_search",
   302→                gatherMethod: "web",
   303→              });
   304→            }
   305→            send("step", {
   306→              step: "gather",
   307→              status: "progress",
   308→              message: `Web search ${wi + 1}/${webSearchQueries.length}: "${webSearchQueries[wi].slice(0, 50)}" → ${searchResults.length} results`,
   309→            });
   310→          } catch {}
   311→          await new Promise((r) => setTimeout(r, 1000));
   312→        }
   313→

   315→        const allItems = [...dbItems, ...webItems];
   316→        const seenExternalIds = new Set<string>();
   317→        const uniqueItems: any[] = [];
   318→        for (const item of allItems) {
   319→          const dedupKey = `${item.source}:${item.externalId || item.url}`;
   320→          if (seenExternalIds.has(dedupKey)) continue;
   321→          seenExternalIds.add(dedupKey);
   322→          uniqueItems.push(item);
   323→        }
   324→        uniqueItems.sort((a, b) => {
   325→          const score = (x: any) =>
   326→            (x.source === "pubmed" ? 4 : 0) +
   327→            (x.source === "rcsb" && x.extra?.hasPublication ? 3 : 0) +
   328→            (x.source === "rcsb" ? 2 : 0) +
   329→            (x.abstract ? 1 : 0);
   330→          return score(b) - score(a);
   331→        });
   332→
   333→        const savedDataSources: any[] = [];
   334→        const savedReferences: any[] = [];
   335→        for (const item of uniqueItems) {
   336→          try {
   337→            const ds = await db.dataSource.create({
   338→              data: {
   339→                projectId,
   340→                source: item.source,
   341→                query: item.queryUsed || item.title,
   342→                rawJson: JSON.stringify({ items: [item] }),
   343→                title: item.title,
   344→                externalId: item.externalId,
   345→                url: item.url,
   346→                authors: item.authors || null,
   347→                journal: item.journal || null,
   348→                year: item.year || null,
   349→                doi: item.doi || null,
   350→                abstract: item.abstract || null,
   351→                extra: item.extra ? JSON.stringify(item.extra) : null,
   352→                pinned: true,
   353→              },
   354→            });

   372→                    title: item.title,
   373→                    authors: item.authors || null,
   374→                    journal: item.journal || null,
   375→                    year: item.year || null,
   376→                    url: item.url || null,
   377→                    doi: item.doi || null,
   378→                    abstract: item.abstract || null,
   379→                    projectId,
   380→                  },

   384→            }
   385→          } catch {}
   386→        }
   387→
   388→        send("step", {
   389→          step: "gather",
   390→          status: "done",

   411→          step: "curate",
   412→          status: "done",
   413→          curatedCount: curatedRefs.length,

   425→TARGET TOTAL WORDS: ${targetWords}

   429→${curatedRefs.slice(0, 30).map((r: any, i: number) => `[${i + 1}] ${r.authors || "Anon"} (${r.year || "n.d."}) ${r.title?.slice(0, 80) || ""}`).join("\n")}

   431→Plan a comprehensive review article with ${Math.max(5, Math.ceil(targetWords / 400))}-${Math.max(8, Math.ceil(targetWords / 300))} sections.
   432→Each section 200-450 words. Sections must cover DISTINCT aspects of the topic.
   433→
   434→Respond as STRICT JSON:
   435→{
   436→  "sections": [
   437→    { "title": "descriptive title", "focus": "what this section covers", "targetWords": 300 }
   438→  ]
   439→}
   440→Output JSON only.`;
   441→
   442→        const planRaw = await chatWithSession(projectId, planPrompt, {
   443→          system: planSystem,
   444→          temperature: 0.5,
   445→          taskType: "plan",
   446→          maxTokens,
   447→          metadata: { step: "plan", targetWords },
   448→        });
   449→        const planParsed = safeParseJSON(planRaw, { sections: [] });
   450→        let sections: any[] = (planParsed.sections || []).filter((s: any) => s.title && s.targetWords);
   451→
   452→        if (sections.length === 0) {
   453→          const fallbackCount = Math.max(5, Math.ceil(targetWords / 300));
   454→          sections = [];
   455→          for (let i = 0; i < fallbackCount; i++) {
   456→            sections.push({
   457→              title: i === 0 ? "Introduction" : i === fallbackCount - 1 ? "Future Directions" : `Section ${i + 1}`,
   458→              targetWords: Math.floor(targetWords / fallbackCount),
   459→              focus: `Aspect ${i + 1} of ${project.topic}`,
   460→            });
   461→          }
   462→        }
   463→        const minSections = Math.max(5, Math.ceil(targetWords / 300));
   464→        if (sections.length < minSections) {
   465→          const perSectionTarget = Math.floor(targetWords / Math.max(sections.length, 1));
   466→          for (const s of sections) s.targetWords = perSectionTarget;
   467→        }
   468→        // Dedup titles
   469→        const seenTitles = new Set<string>();
   470→        for (const s of sections) {
   471→          const tl = (s.title || "").toLowerCase().trim();
   472→          if (seenTitles.has(tl)) s.title = `${s.title} (continued)`;
   473→          seenTitles.add(tl);
   474→          if (!s.focus) s.focus = `Discussion of ${s.title} in the context of ${project.topic}`;
   475→        }
   476→
   477→        send("step", {
   478→          step: "plan",
   479→          status: "done",
   480→          sectionCount: sections.length,
   481→          sections: sections.map((s: any) => ({ title: s.title, targetWords: s.targetWords })),
   482→          message: `Planned ${sections.length} sections.`,
   483→        });
   484→        log(`plan: ${sections.length} sections`);
   485→
   486→        // ============ STEP 4: ★ Analyze — extract evidence bank ============
   487→        send("step", {
   488→          step: "analyze",
   489→          status: "started",
   490→          message: `Analyzing ${curatedRefs.length} sources and extracting an evidence bank (claims pre-bound to references)...`,
   491→        });
   492→
   493→        const evidenceBank = await extractEvidenceBank(
   494→          projectId,
   495→          curatedRefs as EvidenceRefInput[],
   496→          project.topic,
   497→          project.field || "life sciences",
   498→          { maxRefs: Math.min(curatedRefs.length, 40), batchSize: 14, maxTokens }
   499→        );
   500→
   501→        send("step", {
   502→          step: "analyze",
   503→          status: "done",
   504→          evidenceItems: evidenceBank.length,
   505→          refsAnalyzed: Math.min(curatedRefs.length, 40),
   506→          message: `Extracted ${evidenceBank.length} evidence claims from ${Math.min(curatedRefs.length, 40)} references.`,
   507→          detail: evidenceBank.slice(0, 6).map((e) => `[REF-${e.refIndex}] ${e.claim.slice(0, 100)}`).join("\n"),
   508→        });
   509→        log(`analyze: ${evidenceBank.length} evidence items`);
   510→
   511→        // ============ STEP 5: ★ Allocate evidence to sections ============
   512→        send("step", { step: "allocate", status: "started", message: "Allocating references + evidence to sections..." });
   513→
   514→        const allocations = await allocateEvidenceToSections(
   515→          projectId,
   516→          sections,
   517→          curatedRefs as EvidenceRefInput[],
   518→          evidenceBank,
   519→          project.topic,
   520→          { minRefsPerSection: 5, maxRefsPerSection: 12, maxTokens }
   521→        );
   522→
   523→        send("step", {
   524→          step: "allocate",

  1031→          references: globalRefs.length,
  1032→        });
  1033→        log(`compose: article saved (${countWords(articleContent)} words, ${globalRefs.length} refs, audit: ${audit.summary.blockingErrors} blocking, ${audit.summary.suspect + audit.summary.unsupported} topicality warnings)`);
  1034→
  1035→        const totalMs = Date.now() - t0;
  1036→        send("complete", {
  1037→          articleId: article.id,
  1038→          wordCount: countWords(articleContent),
  1039→          references: globalRefs.length,
  1040→          sections: generatedParagraphs.length,
  1041→          totalMs,
  1042→          pipeline: "v2",
  1043→          accuracy: {
  1044→            droppedKeys: stats.droppedKeys,
  1045→            strippedNumeric: stats.strippedNumeric,
  1046→            gateRetries: stats.gateRetries,
  1047→            citationsChecked: stats.citationsChecked,
  1048→            citationsRemoved: stats.citationsRemoved,
  1049→            citationsFlagged: stats.citationsFlagged,
  1050→            auditBlockingErrors: audit.summary.blockingErrors,
  1051→            auditTopicalityWarnings: audit.summary.suspect + audit.summary.unsupported,
  1052→            auditOrphans: audit.summary.orphan,
  1053→          },
  1054→          message: `v2 pipeline complete: ${countWords(articleContent)} words, ${globalRefs.length} references, ${stats.citationsChecked} citations adversarially verified (${stats.citationsRemoved} removed).`,
  1055→        });
  1056→        safeClose();
  1057→      } catch (err: any) {
  1058→        const errMsg = String(err?.message ?? err);
  1059→        log(`FATAL: ${errMsg.slice(0, 300)}`);
  1060→        send("error", { error: `v2 pipeline failed: ${errMsg.slice(0, 300)}` });

  1066→  return new Response(stream, {

  1075→/* ================================================================
  1076→ * Adversarial per-citation verification (used by STEP 7 and by the
  1077→ * standalone /api/articles/[id]/adversarial-review route).
  1078→ * ================================================================ */
  1079→
  1080→export interface VerifyCheck {
  1081→  n: number;
  1082→  sentence: string;
  1083→  verdict: "SUPPORTED" | "UNSUPPORTED" | "PARTIAL";
  1084→  confidence: number;
  1085→  reason: string;
  1086→}
  1087→
  1088→export interface VerifySectionResult {
  1089→  checked: number;
  1090→  removals: VerifyCheck[];
  1091→  flagged: VerifyCheck[];
  1092→  removedNums: number[];
  1093→  llmCalls: number;
  1094→}
  1095→
  1096→export async function adversarialVerifySection(
  1097→  projectId: string,
  1098→  content: string,
  1099→  refs: any[],
  1100→  opts: {
  1101→    batchSize?: number;
  1102→    removeVerdict?: string;
  1103→    removeConfidence?: number;
  1104→    maxTokens?: number;
  1105→  } = {}
  1106→): Promise<VerifySectionResult> {
  1107→  const batchSize = opts.batchSize ?? 10;
  1108→  const removeVerdict = opts.removeVerdict ?? "UNSUPPORTED";
  1109→  const removeConfidence = opts.removeConfidence ?? 80;
  1110→
  1111→  const { body } = splitBodyAndReferences(content);
  1112→  const citations = extractBodyCitations(body).filter((c) => c.n >= 1 && c.n <= refs.length);
  1113→  if (!citations.length) {
  1114→    return { checked: 0, removals: [], flagged: [], removedNums: [], llmCalls: 0 };
  1115→  }
  1116→
  1117→  // De-duplicate by n (multiple sentences citing the same ref are checked once
  1118→  // per DISTINCT sentence, capped at 2 sentences per ref to bound cost)
  1119→  const byRef = new Map<number, { n: number; sentence: string }[]>();
  1120→  for (const c of citations) {
  1121→    const arr = byRef.get(c.n) || [];
  1122→    if (arr.length < 2) arr.push({ n: c.n, sentence: c.sentence });
  1123→    byRef.set(c.n, arr);
  1124→  }
  1125→  const checks: { n: number; sentence: string }[] = [];
  1126→  for (const arr of byRef.values()) checks.push(...arr);
  1127→
  1128→  const system = `You are a rigorous peer reviewer auditing citation accuracy in a scientific review article.
  1129→For each CHECK you receive a claim sentence from the article and the FULL metadata of the reference it cites.
  1130→Your job: decide whether that specific reference actually supports the specific claim.
  1131→
  1132→Be rigorous BUT FAIR:
  1133→- UNSUPPORTED only when the reference is genuinely about a DIFFERENT subject than the claim
  1134→  (different protein/method/organism/topic), or the claim asserts specifics that clearly
  1135→  contradict or are absent from the reference's subject matter.
  1136→- Topical match is enough for SUPPORTED: if the reference's title/abstract covers the SAME
  1137→  subject as the claim, verdict SUPPORTED even if the wording differs or the sentence adds
  1138→  hyperbole ("unprecedented", "significant progress").
  1139→- PARTIAL when the reference covers the topic but the sentence asserts a very specific
  1140→  statistic or mechanism detail you cannot find in the reference.
  1141→- Do NOT verdict UNSUPPORTED while your own reason says the reference "explicitly
  1142→  describes/defines/states" the claim — that is a contradiction. If the reference covers
  1143→  the claim, it is SUPPORTED. Be consistent between your verdict and your reason.
  1144→
  1145→Respond as STRICT JSON only:
  1146→{"checks":[{"id":1,"verdict":"SUPPORTED|UNSUPPORTED|PARTIAL","confidence":0,"reason":"one line"}]}
  1147→confidence is 0-100 (how sure you are of YOUR verdict). Output JSON only.`;
  1148→
  1149→  const removals: VerifyCheck[] = [];
  1150→  const flagged: VerifyCheck[] = [];
  1151→  let llmCalls = 0;
  1152→
  1153→  for (let b = 0; b < checks.length; b += batchSize) {
  1154→    const batch = checks.slice(b, b + batchSize);
  1155→    const block = batch
  1156→      .map((c, i) => {
  1157→        const ref = refs[c.n - 1] || {};
  1158→        const auth = (ref.authors || "Anon").trim();
  1159→        const yr = ref.year ? ` (${ref.year})` : "";
  1160→        const jour = ref.journal ? `, ${ref.journal}` : "";
  1161→        const abs = ref.abstract ? `\n  Abstract: ${ref.abstract.slice(0, 600)}` : "\n  (no abstract)";
  1162→        return `[CHECK ${i + 1}] (cited as [${c.n}])\n  Claim sentence: "${c.sentence.slice(0, 400)}"\n  Reference: ${auth}${yr}${jour}. ${ref.title || "Untitled"}.${abs}`;
  1163→      })
  1164→      .join("\n\n");
  1165→
  1166→    const prompt = `CHECKS:
  1167→${block}
  1168→
  1169→Adjudicate every check. Respond as STRICT JSON:
  1170→{"checks":[{"id":1,"verdict":"...","confidence":0,"reason":"..."}]}`;
  1171→
  1172→    try {
  1173→      llmCalls++;
  1174→      const raw = await chatWithSession(projectId, prompt, {
  1175→        system,
  1176→        temperature: 0.1,
  1177→        taskType: "verify",
  1178→        maxTokens: opts.maxTokens,
  1179→        metadata: { step: "adversarial-verify", batch: Math.floor(b / batchSize) + 1 },
  1180→      });
  1181→      const parsed = safeParseJSON(raw, { checks: [] });
  1182→      for (const c of parsed.checks || []) {
  1183→        const id = parseInt(String(c.id), 10);
  1184→        if (isNaN(id) || id < 1 || id > batch.length) continue;
  1185→        let verdict = String(c.verdict || "").toUpperCase();
  1186→        if (!["SUPPORTED", "UNSUPPORTED", "PARTIAL"].includes(verdict)) continue;
  1187→        const reason = String(c.reason || "").slice(0, 240);
  1188→        // Contradiction guard (E2E finding): verdict=UNSUPPORTED while the
  1189→        // reason says the reference "explicitly describes/defines/states" the
  1190→        // claim is a reviewer false positive — downgrade to PARTIAL (flag only).
  1191→        if (
  1192→          verdict === "UNSUPPORTED" &&
  1193→          (/explicitly (describes|defines|states|discusses|mentions|reports|shows|demonstrates)/i.test(reason) ||
  1194→            /directly matches/i.test(reason) ||
  1195→            /(?:directly|closely) (?:relates|aligns|correspond)s?/i.test(reason))
  1196→        ) {
  1197→          verdict = "PARTIAL";
  1198→        }
  1199→        const confidence = Math.max(0, Math.min(100, parseInt(String(c.confidence ?? 50), 10) || 50));
  1200→        const item: VerifyCheck = {
  1201→          n: batch[id - 1].n,
  1202→          sentence: batch[id - 1].sentence,
  1203→          verdict: verdict as VerifyCheck["verdict"],
  1204→          confidence,
  1205→          reason,
  1206→        };
  1207→        if (verdict === removeVerdict && confidence >= removeConfidence) {
  1208→          removals.push(item);
  1209→        } else if (verdict === "UNSUPPORTED" || verdict === "PARTIAL") {
  1210→          flagged.push(item);
  1211→        }
  1212→      }
  1213→    } catch (err: any) {
  1214→      console.warn(`[adversarialVerifySection] batch failed: ${err?.message?.slice(0, 100)}`);
  1215→    }
  1216→    await new Promise((r) => setTimeout(r, 1500));
  1217→  }
  1218→
  1219→  return {
  1220→    checked: checks.length,
  1221→    removals,
  1222→    flagged,
  1223→    removedNums: [...new Set(removals.map((r) => r.n))],
  1224→    llmCalls,
  1225→  };
  1226→}
