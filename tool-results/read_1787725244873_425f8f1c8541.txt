    69→
    70→interface Props {
    71→  paragraph: Paragraph & { annotations: Annotation[]; references: any[] };
    72→  projectId: string;
    73→  index: number;
    74→  /** The latest composed article content — used to extract global reference
    75→   * data so that hover tooltips resolve [20] etc. correctly. */
    76→  articleContent?: string;
    77→}
    78→
    79→export function ParagraphCard({ paragraph, projectId, index, articleContent }: Props) {
    80→  const { t } = useI18n();
    81→  const qc = useQueryClient();
    82→  const [editing, setEditing] = React.useState(false);
    83→  const [draft, setDraft] = React.useState(paragraph.content);
    84→  const [annOpen, setAnnOpen] = React.useState(false);
    85→  const [activeAnnotation, setActiveAnnotation] = React.useState<Annotation | null>(null);
    86→  const [selection, setSelection] = React.useState<{ text: string; rect: DOMRect } | null>(null);
    87→  const [undoSnapshot, setUndoSnapshot] = React.useState<string | null>(null);
    88→  const [diffOpen, setDiffOpen] = React.useState(false);
    89→  const [validateOpen, setValidateOpen] = React.useState(false);
    90→  const [viewLang, setViewLang] = React.useState<"en" | "zh">("en");
    91→  const bodyRef = React.useRef<HTMLDivElement>(null);
    92→  const pendingMarkRef = React.useRef<HTMLElement | null>(null);
    93→
    94→  // Check if this paragraph has a Chinese translation
    95→  const hasZh = !!(paragraph as any).contentZh;
    96→  // Pick the content to display based on the language toggle
    97→  const displayContent = viewLang === "zh" && hasZh
    98→    ? (paragraph as any).contentZh
    99→    : paragraph.content;
   100→
   101→  // Extract global article references from the composed article's
   102→  // ## References section. This allows hover tooltips to resolve globally
   103→  // numbered citations like [20] even when the paragraph's own references
   104→  // list is local (7 entries).
   105→  const globalArticleRefs = React.useMemo<CitationRef[]>(() => {
   106→    if (!articleContent) return [];
   107→    const refHeaderIdx = articleContent.indexOf("## References");
   108→    const bareRefIdx = articleContent.indexOf("\nREFERENCES\n");
   109→    if (refHeaderIdx >= 0) {
   110→      const refText = articleContent.slice(refHeaderIdx);
   111→      try {
   112→        const parsed = parseCitationsBlock(refText);
   113→        if (parsed.length > 0) return parsed;
   114→      } catch {}
   115→    }
   116→    if (bareRefIdx >= 0) {
   117→      const refText = articleContent.slice(bareRefIdx + 1);
   118→      try {
   119→        const parsed = parseCitationsBlock(refText);
   120→        if (parsed.length > 0) return parsed;
   121→      } catch {}
   122→    }
   123→    return [];
   124→  }, [articleContent]);
   125→
   126→  // Resolve citations against the PARAGRAPH'S OWN reference list. The
   127→  // paragraph body carries its own numbering ([n] = the n-th reference
   128→  // attached to this paragraph, ordered by citationOrder), so local refs are
   129→  // the only correct resolver here. Using the composed article's global
   130→  // "## References" list caused out-of-range markers (e.g. [14] vs a 13-ref
   131→  // article) to render as red "?" chips.
   132→  // The global list is only a fallback for paragraphs with no refs attached.
   133→  const effectiveRefs =
   134→    (paragraph.references?.length ?? 0) > 0
   135→      ? paragraph.references
   136→      : globalArticleRefs;
   137→
   138→  const invalidate = () => qc.invalidateQueries({ queryKey: ["project", projectId] });
   139→
   140→  const updateMut = useMutation({
   141→    mutationFn: (input: Partial<Paragraph>) =>
   142→      api.updateParagraph(paragraph.id, input),
   143→    onSuccess: () => {
   144→      toast.success(t("toast.paragraphUpdated"));
   145→      setEditing(false);
   146→      invalidate();
   147→    },
   148→    onError: (e: Error) => toast.error(e.message),
   149→  });
   150→
   151→  const deleteMut = useMutation({
   152→    mutationFn: () => api.deleteParagraph(paragraph.id),
   153→    onSuccess: () => {
   154→      toast.success(t("toast.paragraphDeleted"));
   155→      invalidate();
   156→    },
   157→    onError: (e: Error) => toast.error(e.message),
   158→  });
   159→
   160→  const addAnnMut = useMutation({
   161→    mutationFn: (input: Partial<Annotation>) =>
   162→      api.addAnnotation(paragraph.id, input),
   163→    onSuccess: () => {
   164→      toast.success(t("toast.annotationAdded"));
   165→      clearSelection();
   166→      invalidate();
   167→    },
   168→    onError: (e: Error) => toast.error(e.message),
   169→  });
   170→
   171→  const resolveAnnMut = useMutation({
   172→    mutationFn: ({ id, resolved }: { id: string; resolved: boolean }) =>
   173→      api.updateAnnotation(id, { resolved }),
   174→    onSuccess: () => invalidate(),
   175→    onError: (e: Error) => toast.error(e.message),
   176→  });
   177→
   178→  const deleteAnnMut = useMutation({
   179→    mutationFn: (id: string) => api.deleteAnnotation(id),
   180→    onSuccess: () => {
   181→      toast.success(t("toast.annotationRemoved"));
   182→      invalidate();
   183→    },
   184→    onError: (e: Error) => toast.error(e.message),
   185→  });
   186→
   187→  const reviseMut = useMutation({
   188→    mutationFn: async (input: { mode?: string; instructions?: string }) => {
   189→      // Save snapshot before revising (for undo)
   190→      setUndoSnapshot(paragraph.content);
   191→      return api.reviseParagraph(paragraph.id, input);
   192→    },
   193→    onSuccess: () => {
   194→      toast.success(t("toast.paragraphRevised"));
   195→      invalidate();
   196→    },
   197→    onError: (e: Error) => {
   198→      setUndoSnapshot(null);
   199→      toast.error(e.message);
   200→    },
   201→  });
   202→
   203→  const regenerateMut = useMutation({
   204→    mutationFn: async () => {
   205→      const res = await fetch(`/api/paragraphs/${paragraph.id}/regenerate`, {
   206→        method: "POST",
   207→        headers: { "Content-Type": "application/json" },
   208→      });
   209→      if (!res.ok) {
   210→        const txt = await res.text();
   211→        throw new Error(txt || `Regenerate failed (${res.status})`);
   212→      }
   213→      return res.json();
   214→    },
   215→    onSuccess: () => {
   216→      toast.success(t("toast.paragraphRegenerated") || "Section regenerated.");
   217→      invalidate();
   218→    },
   219→    onError: (e: Error) => toast.error(e.message),
   220→  });
   221→
   222→  const undoReviseMut = useMutation({
   223→    mutationFn: async () => {
   224→      if (!undoSnapshot) throw new Error("No undo snapshot available.");
   225→      return api.updateParagraph(paragraph.id, {
   226→        content: undoSnapshot,
   227→        status: "annotated",
   228→      });
   229→    },
   230→    onSuccess: () => {
   231→      toast.success(t("toast.paragraphReverted"));
   232→      setUndoSnapshot(null);
   233→      invalidate();
   234→    },
   235→    onError: (e: Error) => toast.error(e.message),
   236→  });
   237→
   238→  // Unwrap any pending-selection <mark> element (restore original DOM)
   239→  const unwrapPendingMark = React.useCallback(() => {
   240→    const mark = pendingMarkRef.current;
   241→    if (!mark || !mark.parentNode) {
   242→      pendingMarkRef.current = null;
   243→      return;
   244→    }
   245→    const parent = mark.parentNode;
   246→    // Move all children out of the mark, then remove it
   247→    while (mark.firstChild) {
   248→      parent.insertBefore(mark.firstChild, mark);
   249→    }
   250→    parent.removeChild(mark);
   251→    // Normalize adjacent text nodes
   252→    try { parent.normalize(); } catch {}
   253→    pendingMarkRef.current = null;
   254→  }, []);
   255→
   256→  // Capture text selection within this paragraph body
   257→  const handleMouseUp = React.useCallback(() => {
   258→    const sel = window.getSelection();
   259→    if (!sel || sel.isCollapsed || sel.rangeCount === 0) {
   260→      return;
   261→    }
   262→    const text = sel.toString().trim();
   263→    if (text.length < 2) {
   264→      return;
   265→    }
   266→    const range = sel.getRangeAt(0);
   267→    const bodyEl = bodyRef.current;
   268→    if (!bodyEl || !bodyEl.contains(range.commonAncestorContainer)) {
   269→      return;
   270→    }
   271→    const rect = range.getBoundingClientRect();
   272→
   273→    // Unwrap any previous pending mark
   274→    unwrapPendingMark();
   275→
   276→    // Wrap the selected range in a <mark class="pending-selection"> so the
   277→    // highlight persists even after the native selection collapses (which
   278→    // happens when the Popover steals focus).
   279→    try {
   280→      const contents = range.extractContents();
   281→      const mark = document.createElement("mark");
   282→      mark.className = "pending-selection";
   283→      mark.appendChild(contents);
   284→      range.insertNode(mark);
   285→      pendingMarkRef.current = mark;
   286→    } catch {
   287→      // If DOM manipulation fails, the selection toolbar still works —
   288→      // just without the persistent highlight
   289→    }
   290→
   291→    // Clear the native selection (it will be lost anyway when popover opens)
   292→    sel.removeAllRanges();
   293→
   294→    setSelection({ text, rect });
   295→  }, [unwrapPendingMark]);
   296→
   297→  // Cleanup pending mark on unmount
   298→  React.useEffect(() => {
   299→    return () => unwrapPendingMark();
   300→  }, [unwrapPendingMark]);
   301→
   302→  // Clear selection + unwrap the pending highlight mark
   303→  const clearSelection = React.useCallback(() => {
   304→    unwrapPendingMark();
   305→    setSelection(null);
   306→  }, [unwrapPendingMark]);
   307→
   308→  const unresolvedCount = paragraph.annotations.filter((a) => !a.resolved).length;
   309→  const status = STATUS_STYLES[paragraph.status] || STATUS_STYLES.draft;
   310→  const formatMeta = PARAGRAPH_FORMATS.find((f) => f.id === paragraph.format);
   311→
   312→  const ANN_CARD_CLASS: Record<string, string> = {
   313→    emerald: "border-emerald-200 bg-emerald-50/50 dark:border-emerald-900/50 dark:bg-emerald-950/20",
   314→    teal: "border-teal-200 bg-teal-50/50 dark:border-teal-900/50 dark:bg-teal-950/20",
   315→    amber: "border-amber-200 bg-amber-50/50 dark:border-amber-900/50 dark:bg-amber-950/20",
   316→    rose: "border-rose-200 bg-rose-50/50 dark:border-rose-900/50 dark:bg-rose-950/20",
   317→    violet: "border-violet-200 bg-violet-50/50 dark:border-violet-900/50 dark:bg-violet-950/20",
   318→    sky: "border-sky-200 bg-sky-50/50 dark:border-sky-900/50 dark:bg-sky-950/20",
   319→  };
   320→
   321→  return (
   322→    <div className="rounded-xl border border-border/70 bg-card shadow-sm hover:shadow-md transition-shadow overflow-hidden acad-fade-in">
   323→      {/* Header */}
   324→      <div className="flex items-start gap-3 px-4 py-3 border-b border-border/50 bg-gradient-to-r from-muted/40 to-transparent">
   325→        <span className="text-[10px] font-mono text-muted-foreground mt-1 shrink-0">
   326→          §{String(index + 1).padStart(2, "0")}
   327→        </span>
   328→        <div className="flex-1 min-w-0">
   329→          {editing ? (
   330→            <input
   331→              defaultValue={paragraph.title}
   332→              onBlur={(e) =>
   333→                updateMut.mutate({ title: e.target.value })
   334→              }
   335→              className="text-sm font-semibold bg-transparent border-b border-dashed border-primary/40 focus:outline-none w-full"
   336→            />
   337→          ) : (
   338→            <h3 className="text-sm font-semibold leading-tight truncate">
   339→              {paragraph.title}
   340→            </h3>
   341→          )}
   342→          <div className="flex flex-wrap items-center gap-1 mt-1.5">
   343→            <span
   344→              className={`badge-${status.color} inline-flex items-center gap-1 px-1.5 py-0.5 rounded text-[9px] font-semibold uppercase tracking-wide`}
   345→            >
   346→              <Icon name={status.icon} className="h-2.5 w-2.5" />
   347→              {status.label}
   348→            </span>
   349→            <span className="text-[9px] text-muted-foreground">
   350→              {formatMeta?.label || paragraph.format}
   351→            </span>
   352→            <span className="text-[9px] text-muted-foreground">·</span>
   353→            <span className="text-[9px] text-muted-foreground">
   354→              {viewLang === "zh" && hasZh
   355→                ? `${(paragraph as any).wordCountZh || 0}字`
   356→                : t("para.wordsCount", { n: paragraph.wordCount })}
   357→            </span>
   358→            {/* v101-4: Citation count badge */}
   359→            {(() => {
   360→              const citCount = (displayContent?.match(/\[\d+(?:[,\-–\s]\d+)*\]/g) || []).length;
   361→              if (citCount > 0) {
   362→                return (
   363→                  <span className="inline-flex items-center gap-0.5 text-[9px] text-amber-600 dark:text-amber-400 font-medium bg-amber-500/10 px-1.5 py-0.5 rounded">
   364→                    <Quote className="h-2.5 w-2.5" />
   365→                    {citCount} cit
   366→                  </span>
   367→                );
   368→              }
   369→              return null;
   370→            })()}
   371→            {hasZh && (
   372→              <span className="text-[9px] text-fuchsia-600 dark:text-fuchsia-400 font-medium">
   373→                · 中文
   374→              </span>
   375→            )}
   376→            {unresolvedCount > 0 && (
   377→              <span className="inline-flex items-center gap-0.5 text-[9px] text-amber-600 dark:text-amber-400 font-medium">
   378→                <MessageSquare className="h-2.5 w-2.5" />
   379→                {unresolvedCount}
   380→              </span>
   381→            )}
   382→            {/* EN/ZH toggle — only when the paragraph has a Chinese translation */}
   383→            {hasZh && !editing && (
   384→              <div className="ml-auto flex items-center gap-0.5 rounded-md border border-border/40 bg-muted/30 p-0.5">
   385→                <button
   386→                  type="button"
   387→                  onClick={() => setViewLang("en")}
   388→                  className={`px-1.5 py-0.5 rounded text-[9px] font-semibold transition-colors ${
   389→                    viewLang === "en"
   390→                      ? "bg-primary text-primary-foreground shadow-sm"
   391→                      : "text-muted-foreground hover:text-foreground"
   392→                  }`}
   393→                  title="English"
   394→                >
   395→                  EN
   396→                </button>
   397→                <button
   398→                  type="button"
   399→                  onClick={() => setViewLang("zh")}
   400→                  className={`px-1.5 py-0.5 rounded text-[9px] font-semibold transition-colors ${
   401→                    viewLang === "zh"
   402→                      ? "bg-primary text-primary-foreground shadow-sm"
   403→                      : "text-muted-foreground hover:text-foreground"
   404→                  }`}
   405→                  title="中文"
   406→                >
   407→                  中
   408→                </button>
   409→              </div>
   410→            )}
   411→          </div>
   412→        </div>
   413→        <DropdownMenu>
   414→          <DropdownMenuTrigger asChild>
   415→            <Button variant="ghost" size="icon" className="h-7 w-7 shrink-0">
   416→              <MoreHorizontal className="h-3.5 w-3.5" />
   417→            </Button>
   418→          </DropdownMenuTrigger>
   419→          <DropdownMenuContent align="end" className="w-44">
   420→            <DropdownMenuItem onClick={() => setEditing((v) => !v)}>
   421→              <Pencil className="h-3.5 w-3.5" /> {editing ? t("para.stopEditing") : t("para.editContent")}
   422→            </DropdownMenuItem>
   423→            <DropdownMenuItem
   424→              onClick={() =>
   425→                navigator.clipboard.writeText(paragraph.content).then(() =>
   426→                  toast.success(t("toast.copiedToClipboard"))
   427→                )
   428→              }
   429→            >
   430→              <Copy className="h-3.5 w-3.5" /> {t("para.copyText")}
   431→            </DropdownMenuItem>
   432→            <DropdownMenuItem onClick={() => setValidateOpen(true)}>
   433→              <ShieldCheck className="h-3.5 w-3.5" /> {t("para.validateCitations")}
   434→            </DropdownMenuItem>
   435→            <DropdownMenuItem
   436→              onClick={() => regenerateMut.mutate()}
   437→              disabled={regenerateMut.isPending}
   438→            >
   439→              {regenerateMut.isPending ? (
   440→                <Loader2 className="h-3.5 w-3.5 animate-spin" />
   441→              ) : (
   442→                <RotateCw className="h-3.5 w-3.5" />
   443→              )}
   444→              {t("para.regenerate") || "Regenerate section"}
   445→            </DropdownMenuItem>
   446→            <DropdownMenuSeparator />
   447→            <FormatSelect
   448→              value={paragraph.format}
   449→              onChange={(v) => updateMut.mutate({ format: v })}
   450→              options={PARAGRAPH_FORMATS.map((f) => ({ value: f.id, label: f.label }))}
   451→            />
   452→            <FormatSelect
   453→              value={paragraph.scenario}
   454→              onChange={(v) => updateMut.mutate({ scenario: v })}
   455→              options={PARAGRAPH_SCENARIOS.map((s) => ({ value: s.id, label: s.label }))}
   456→            />
   457→            <DropdownMenuSeparator />
   458→            <DropdownMenuItem
   459→              className="text-destructive focus:text-destructive"
   460→              onClick={() => deleteMut.mutate()}
   461→            >
   462→              <Trash2 className="h-3.5 w-3.5" /> {t("para.delete")}
   463→            </DropdownMenuItem>
   464→          </DropdownMenuContent>
   465→        </DropdownMenu>
   466→      </div>
   467→
   468→      {/* Body */}
   469→      <div className="px-4 py-3 relative paper-surface" ref={bodyRef} onMouseUp={handleMouseUp}>
   470→        {editing ? (
   471→          <div className="space-y-2">
   472→            <Textarea
   473→              value={draft}
   474→              onChange={(e) => setDraft(e.target.value)}
   475→              className="font-serif-text text-sm min-h-[160px] leading-relaxed"
   476→            />
   477→            <div className="flex items-center gap-2 flex-wrap">
   478→              <Button
   479→                size="sm"
   480→                onClick={() => updateMut.mutate({ content: draft })}
   481→                disabled={updateMut.isPending}
   482→              >
   483→                {updateMut.isPending && <Loader2 className="h-3.5 w-3.5 animate-spin" />}
   484→                {t("common.save")}
   485→              </Button>
   486→              <Button
   487→                size="sm"
   488→                variant="ghost"
   489→                onClick={() => {
   490→                  setDraft(paragraph.content);
   491→                  setEditing(false);
   492→                }}
   493→              >
   494→                {t("common.cancel")}
   495→              </Button>
   496→              <InsertStructureAnalysisButton
   497→                projectId={projectId}
   498→                onInsert={(markdown) => {
   499→                  setDraft((d) => d + (d && !d.endsWith("\n") ? "\n\n" : "") + markdown);
   500→                  toast.success(t("structure.insertedAnalysis"));
   501→                }}
   502→              />
   503→            </div>
   504→          </div>
   505→        ) : (
   506→          <MarkdownCitations
   507→            content={displayContent}
   508→            annotations={paragraph.annotations}
   509→            references={effectiveRefs}
   510→            onAnnotationClick={(a) => setActiveAnnotation(a)}
   511→          />
   512→        )}
   513→
   514→        {/* Floating selection toolbar */}
   515→        {selection && !editing && (
   516→          <SelectionToolbar
   517→            text={selection.text}
   518→            onSubmit={(comment, type, severity) => {
   519→              const startOffset = paragraph.content.indexOf(selection.text);
   520→              addAnnMut.mutate({
   521→                selectedText: selection.text,
   522→                startOffset: startOffset >= 0 ? startOffset : 0,
   523→                endOffset: startOffset >= 0 ? startOffset + selection.text.length : 0,
   524→                comment,
   525→                type,
   526→                severity,
   527→              });
   528→            }}
   529→            onClose={() => clearSelection()}
   530→            pending={addAnnMut.isPending}
   531→          />
   532→        )}
   533→      </div>
   534→
   535→      {/* Footer: annotations + revise */}
   536→      {paragraph.annotations.length > 0 && (
   537→        <Collapsible open={annOpen} onOpenChange={setAnnOpen} className="border-t border-border/50">
   538→          <div className="px-4 py-2 flex items-center justify-between bg-muted/30">
   539→            <CollapsibleTrigger asChild>
   540→              <Button variant="ghost" size="sm" className="h-7 text-xs gap-1.5">
   541→                <MessageSquare className="h-3.5 w-3.5" />
   542→                {t("para.annotationsCount", { n: paragraph.annotations.length })}
   543→                {annOpen ? (
   544→                  <ChevronUp className="h-3 w-3" />
   545→                ) : (
   546→                  <ChevronDown className="h-3 w-3" />
   547→                )}
   548→              </Button>

   925→/**
   926→ * InsertStructureAnalysisButton — a popover that lists all analyzed RCSB
   927→ * structures in the current project and inserts a compact markdown summary of
   928→ * the selected structure's key metrics into the paragraph draft. This is the
   929→ * "quick insert" counterpart to the full ProteinStructureAnalysisDialog —
   930→ * instead of viewing all 12 tabs, the user picks a structure and gets a
   931→ * one-paragraph summary they can edit into their prose.
   932→ */
   933→function InsertStructureAnalysisButton({
   934→  projectId,
   935→  onInsert,
   936→}: {
   937→  projectId: string;
   938→  onInsert: (markdown: string) => void;
   939→}) {
   940→  const { t } = useI18n();
   941→  const [open, setOpen] = React.useState(false);
   942→  const [loading, setLoading] = React.useState(false);
   943→  const [analyses, setAnalyses] = React.useState<any[]>([]);
   944→
   945→  React.useEffect(() => {
   946→    if (!open) return;
   947→    setLoading(true);
   948→    api
   949→      .listProjectStructures(projectId)
   950→      .then((res) => setAnalyses(res.analyses || []))
   951→      .catch(() => setAnalyses([]))
   952→      .finally(() => setLoading(false));
   953→  }, [open, projectId]);
   954→
   955→  function buildInsertMarkdown(a: any): string {
   956→    // Parse the contextMarkdown to extract key metrics, or build a compact
   957→    // summary from the fields we have.
   958→    const lines: string[] = [];
   959→    lines.push(`> **Structure PDB:${a.pdbId}** — ${a.title || "untitled"}`);
   960→    lines.push(`> - ${a.chainCount} chain(s) · ${a.residueCount} residues · ${a.atomCount} atoms${a.ligandCount > 0 ? ` · ${a.ligandCount} ligand(s)` : ""}`);
   961→    // Try to extract a few key lines from the full context markdown.
   962→    const md = a.contextMarkdown || "";
   963→    const extract = (re: RegExp) => {
   964→      const m = md.match(re);
   965→      return m ? m[1] : null;
   966→    };
   967→    const method = extract(/Method: ([^|]+)/);
   968→    const resolution = extract(/Resolution: ([^|Å]+Å)/);
   969→    const ramaFav = extract(/Favoured \(core\+allowed\): (\S+)%/);
   970→    const ramaOut = extract(/Outliers: (\S+)%/);
   971→    const bfMean = extract(/Mean: ([\d.]+)/);
   972→    const sasaExp = extract(/Exposed: (\S+)%/);
   973→    const sasaBur = extract(/Buried: (\S+)%/);
   974→    const hbonds = extract(/Hydrogen bonds[^:]*: (\d+)/);
   975→    const netCharge = extract(/Net charge at pH 7: ([\d.\-+]+)/);
   976→    const pI = extract(/Isoelectric point \(pI\): ([\d.]+)/);
   977→    const meta: string[] = [];
   978→    if (method) meta.push(`Method: ${method.trim()}`);
   979→    if (resolution) meta.push(`Resolution: ${resolution.trim()}`);
   980→    if (ramaFav) meta.push(`Ramachandran: ${ramaFav}% favoured / ${ramaOut}% outliers`);
   981→    if (bfMean) meta.push(`B̄=${bfMean}`);
   982→    if (sasaExp) meta.push(`SASA: ${sasaExp}% exposed / ${sasaBur}% buried`);
   983→    if (hbonds) meta.push(`${hbonds} H-bonds`);
   984→    if (netCharge) meta.push(`net charge ${netCharge} (pH 7)`);
   985→    if (pI) meta.push(`pI=${pI}`);
   986→    if (meta.length) lines.push(`> - ${meta.join(" · ")}`);
   987→    // Ligands
   988→    const ligMatch = md.match(/Ligands & Cofators \((\d+) detected\):([\s\S]*?)(?=\n\*\*|$)/);
   989→    if (ligMatch && ligMatch[1] !== "0") {
   990→      const ligLines = ligMatch[2].split("\n").filter((l) => l.trim().startsWith("-")).slice(0, 6);
   991→      if (ligLines.length) lines.push(`> - Ligands: ${ligLines.map((l) => l.replace(/^\s*-\s*/, "").replace(/—.*$/, "").trim()).join(", ")}`);
   992→    }
   993→    lines.push(`> _All metrics computed from the actual PDB file via Molcraft structure analysis._`);
   994→    return lines.join("\n");
   995→  }
   996→
   997→  return (
   998→    <Popover open={open} onOpenChange={setOpen}>
   999→      <PopoverTrigger asChild>
  1000→        <Button
  1001→          size="sm"
  1002→          variant="outline"
  1003→          className="h-7 gap-1 text-[10px] border-amber-300 text-amber-700 hover:bg-amber-50 dark:border-amber-800 dark:text-amber-400 dark:hover:bg-amber-950/30"
  1004→          title={t("structure.insertAnalysisTitle")}
  1005→        >
  1006→          <Box className="h-3 w-3" />
  1007→          {t("structure.insertAnalysis")}
  1008→        </Button>
  1009→      </PopoverTrigger>
  1010→      <PopoverContent
  1011→        align="start"
  1012→        className="w-80 p-2 max-h-80 overflow-y-auto"
  1013→      >
  1014→        {loading ? (
  1015→          <div className="flex items-center justify-center py-4">
  1016→            <Loader2 className="h-4 w-4 animate-spin text-muted-foreground" />
  1017→          </div>
  1018→        ) : analyses.length === 0 ? (
  1019→          <div className="text-center py-4 text-[11px] text-muted-foreground">
  1020→            <Box className="h-6 w-6 mx-auto mb-1 opacity-30" />
  1021→            {t("structure.noAnalyzedSources")}
  1022→          </div>
  1023→        ) : (
  1024→          <div className="space-y-1">
  1025→            <div className="text-[10px] uppercase tracking-wider text-muted-foreground font-semibold px-1 pb-1">
  1026→              {t("structure.chooseStructure")}
  1027→            </div>
  1028→            {analyses.map((a) => (
  1029→              <button
  1030→                key={a.pdbId}
  1031→                onClick={() => {
  1032→                  onInsert(buildInsertMarkdown(a));
  1033→                  setOpen(false);
  1034→                }}
  1035→                className="w-full text-left rounded-md border border-border/60 hover:border-amber-300 hover:bg-amber-50/50 dark:hover:bg-amber-950/20 p-2 transition-colors"
  1036→              >
  1037→                <div className="flex items-center gap-1.5">
  1038→                  <Badge
  1039→                    variant="secondary"
  1040→                    className="text-[9px] font-mono bg-amber-100 text-amber-900 dark:bg-amber-950 dark:text-amber-200"
  1041→                  >
  1042→                    {a.pdbId}
  1043→                  </Badge>

  1046→                  </span>
