    39→    39→/**
    40→    40→ * generate-full-v2 — Evidence-grounded article generation pipeline.
    41→    41→ *
    42→    42→ * Multi-stage architecture (deepseek-harness-inspired: analyze → allocate →
    43→    43→ * write, with a validation gate between every stage):
    44→    44→ *
    45→    45→ *   1. gather      — fresh multi-database + web retrieval (same as v1)
    46→    46→ *   2. curate      — LLM selects the most relevant citable subset
    47→    47→ *   3. plan        — LLM designs the section outline
    48→    48→ *   4. analyze     — ★ NEW: extract a structured EVIDENCE BANK (claims
    49→    49→ *                    pre-bound to their sources) from every curated ref
    50→    50→ *   5. allocate    — ★ NEW: assign references + evidence to sections
    51→    51→ *   6. generate    — ★ per-section writing with STRUCTURAL citation keys
    52→    52→ *                    ({{Rn}}) — the LLM never writes numbers, so numbering
    53→    53→ *                    cannot drift; a validation gate retries sections that
    54→    54→ *                    leak raw numeric markers
    55→    55→ *   7. verify      — ★ NEW: adversarial per-citation verification (does this
    56→    56→ *                    specific reference support this specific claim?) with
    57→    57→ *                    conservative removal of unsupported citations
    58→    58→ *   8. compose     — global renumbering + reference sync + article save
    59→    59→ *
    60→    60→ * Accuracy contract (what v2 guarantees that v1 cannot):
    61→    61→ *   - every [n] in the final article was produced by CODE from a {{Rn}} key
    62→    62→ *     that the model copied from a specific reference entry
    63→    63→ *   - every surviving citation survived an adversarial claim-level check
    64→    64→ *     against the reference's own title/abstract
    65→    65→ *   - paragraph reference lists are synced to global numbering (no drift)
    66→    66→ */

  1080→  1080→export interface VerifyCheck {

  1082→  1082→  sentence: string;
  1083→  1083→  verdict: "SUPPORTED" | "UNSUPPORTED" | "PARTIAL";
  1084→  1084→  confidence: number;
  1085→  1085→  reason: string;
  1086→  1086→}

  1088→  1088→export interface VerifySectionResult {
  1089→  1089→  checked: number;
  1090→  1090→  removals: VerifyCheck[];
  1091→  1091→  flagged: VerifyCheck[];
  1092→  1092→  removedNums: number[];
  1093→  1093→  llmCalls: number;
  1094→  1094→}
  1095→  1095→
  1096→  1096→export async function adversarialVerifySection(
  1097→  1097→  projectId: string,
  1098→  1098→  content: string,
  1099→  1099→  refs: any[],
  1100→  1100→  opts: {
  1101→  1101→    batchSize?: number;
  1102→  1102→    removeVerdict?: string;
  1103→  1103→    removeConfidence?: number;
  1104→  1104→    maxTokens?: number;
  1105→  1105→  } = {}
  1106→  1106→): Promise<VerifySectionResult> {
  1107→  1107→  const batchSize = opts.batchSize ?? 10;
  1108→  1108→  const removeVerdict = opts.removeVerdict ?? "UNSUPPORTED";
  1109→  1109→  const removeConfidence = opts.removeConfidence ?? 80;
  1110→  1110→
  1111→  1111→  const { body } = splitBodyAndReferences(content);
  1112→  1112→  const citations = extractBodyCitations(body).filter((c) => c.n >= 1 && c.n <= refs.length);
  1113→  1113→  if (!citations.length) {
  1114→  1114→    return { checked: 0, removals: [], flagged: [], removedNums: [], llmCalls: 0 };
  1115→  1115→  }
  1116→  1116→
  1117→  1117→  // De-duplicate by n (multiple sentences citing the same ref are checked once
  1118→  1118→  // per DISTINCT sentence, capped at 2 sentences per ref to bound cost)
  1119→  1119→  const byRef = new Map<number, { n: number; sentence: string }[]>();
  1120→  1120→  for (const c of citations) {
  1121→  1121→    const arr = byRef.get(c.n) || [];
  1122→  1122→    if (arr.length < 2) arr.push({ n: c.n, sentence: c.sentence });
  1123→  1123→    byRef.set(c.n, arr);
  1124→  1124→  }
  1125→  1125→  const checks: { n: number; sentence: string }[] = [];
  1126→  1126→  for (const arr of byRef.values()) checks.push(...arr);
  1127→  1127→
  1128→  1128→  const system = `You are a rigorous peer reviewer auditing citation accuracy in a scientific review article.
  1129→  1129→For each CHECK you receive a claim sentence from the article and the FULL metadata of the reference it cites.
  1130→  1130→Your job: decide whether that specific reference actually supports the specific claim.
  1131→  1131→
  1132→  1132→Be rigorous BUT FAIR:
  1133→  1133→- UNSUPPORTED only when the reference is genuinely about a DIFFERENT subject than the claim
  1134→  1134→  (different protein/method/organism/topic), or the claim asserts specifics that clearly
  1135→  1135→  contradict or are absent from the reference's subject matter.
  1136→  1136→- Topical match is enough for SUPPORTED: if the reference's title/abstract covers the SAME
  1137→  1137→  subject as the claim, verdict SUPPORTED even if the wording differs or the sentence adds
  1138→  1138→  hyperbole ("unprecedented", "significant progress").
  1139→  1139→- PARTIAL when the reference covers the topic but the sentence asserts a very specific
  1140→  1140→  statistic or mechanism detail you cannot find in the reference.
  1141→  1141→- Do NOT verdict UNSUPPORTED while your own reason says the reference "explicitly
  1142→  1142→  describes/defines/states" the claim — that is a contradiction. If the reference covers
  1143→  1143→  the claim, it is SUPPORTED. Be consistent between your verdict and your reason.
  1144→  1144→
  1145→  1145→Respond as STRICT JSON only:
  1146→  1146→{"checks":[{"id":1,"verdict":"SUPPORTED|UNSUPPORTED|PARTIAL","confidence":0,"reason":"one line"}]}
  1147→  1147→confidence is 0-100 (how sure you are of YOUR verdict). Output JSON only.`;
  1148→  1148→

  1150→  1150→  const flagged: VerifyCheck[] = [];

  1152→  1152→

  1154→  1154→    const batch = checks.slice(b, b + batchSize);
  1155→  1155→    const block = batch
  1156→  1156→      .map((c, i) => {
  1157→  1157→        const ref = refs[c.n - 1] || {};
  1158→  1158→        const auth = (ref.authors || "Anon").trim();
  1159→  1159→        const yr = ref.year ? ` (${ref.year})` : "";
  1160→  1160→        const jour = ref.journal ? `, ${ref.journal}` : "";
  1161→  1161→        const abs = ref.abstract ? `\n  Abstract: ${ref.abstract.slice(0, 600)}` : "\n  (no abstract)";
  1162→  1162→        return `[CHECK ${i + 1}] (cited as [${c.n}])\n  Claim sentence: "${c.sentence.slice(0, 400)}"\n  Reference: ${auth}${yr}${jour}. ${ref.title || "Untitled"}.${abs}`;
  1163→  1163→      })
  1164→  1164→      .join("\n\n");
  1165→  1165→
  1166→  1166→    const prompt = `CHECKS:
  1167→  1167→${block}
  1168→  1168→
  1169→  1169→Adjudicate every check. Respond as STRICT JSON:
  1170→  1170→{"checks":[{"id":1,"verdict":"...","confidence":0,"reason":"..."}]}`;

  1173→  1173→      llmCalls++;

  1175→  1175→        system,
  1176→  1176→        temperature: 0.1,
  1177→  1177→        taskType: "verify",
  1178→  1178→        maxTokens: opts.maxTokens,
  1179→  1179→        metadata: { step: "adversarial-verify", batch: Math.floor(b / batchSize) + 1 },
  1180→  1180→      });

  1185→  1185→        let verdict = String(c.verdict || "").toUpperCase();
  1186→  1186→        if (!["SUPPORTED", "UNSUPPORTED", "PARTIAL"].includes(verdict)) continue;
  1187→  1187→        const reason = String(c.reason || "").slice(0, 240);
  1188→  1188→        // Contradiction guard (E2E finding): verdict=UNSUPPORTED while the
  1189→  1189→        // reason says the reference "explicitly describes/defines/states" the
  1190→  1190→        // claim is a reviewer false positive — downgrade to PARTIAL (flag only).
  1191→  1191→        if (
  1192→  1192→          verdict === "UNSUPPORTED" &&
  1193→  1193→          (/explicitly (describes|defines|states|discusses|mentions|reports|shows|demonstrates)/i.test(reason) ||
  1194→  1194→            /directly matches/i.test(reason) ||
  1195→  1195→            /(?:directly|closely) (?:relates|aligns|correspond)s?/i.test(reason))
  1196→  1196→        ) {
  1197→  1197→          verdict = "PARTIAL";
  1198→  1198→        }
  1199→  1199→        const confidence = Math.max(0, Math.min(100, parseInt(String(c.confidence ?? 50), 10) || 50));
  1200→  1200→        const item: VerifyCheck = {
  1201→  1201→          n: batch[id - 1].n,
  1202→  1202→          sentence: batch[id - 1].sentence,
  1203→  1203→          verdict: verdict as VerifyCheck["verdict"],
  1204→  1204→          confidence,
  1205→  1205→          reason,
  1206→  1206→        };

  1208→  1208→          removals.push(item);
  1209→  1209→        } else if (verdict === "UNSUPPORTED" || verdict === "PARTIAL") {
  1210→  1210→          flagged.push(item);
  1211→  1211→        }
  1212→  1212→      }
  1213→  1213→    } catch (err: any) {
  1214→  1214→      console.warn(`[adversarialVerifySection] batch failed: ${err?.message?.slice(0, 100)}`);
  1215→  1215→    }
  1216→  1216→    await new Promise((r) => setTimeout(r, 1500));
  1217→  1217→  }
  1218→  1218→
  1219→  1219→  return {
  1220→  1220→    checked: checks.length,
  1221→  1221→    removals,
  1222→  1222→    flagged,
  1223→  1223→    removedNums: [...new Set(removals.map((r) => r.n))],
  1224→  1224→    llmCalls,
  1225→  1225→  };
  1226→  1226→}
  1227→  1227→
