    48→function buildHighlightRanges(
    49→  content: string,
    50→  annotations: Annotation[]
    51→): HighlightRange[] {
    52→  const ranges: HighlightRange[] = [];
    53→  for (const a of annotations) {
    54→    if (!a.selectedText || a.resolved) continue;
    55→    const needle = a.selectedText;
    56→    let start = a.startOffset ?? -1;
    57→    if (start < 0 || content.slice(start, start + needle.length) !== needle) {
    58→      start = content.indexOf(needle);
    59→    }
    60→    if (start >= 0) {
    61→      ranges.push({ start, end: start + needle.length, annotation: a });
    62→    }
    63→  }
    64→  const prio: Record<string, number> = { critical: 0, warning: 1, info: 2 };
    65→  ranges.sort(
    66→    (a, b) =>
    67→      a.start - b.start ||
    68→      (prio[a.annotation.severity] ?? 2) - (prio[b.annotation.severity] ?? 2)
    69→  );
    70→  const out: HighlightRange[] = [];
    71→  let lastEnd = -1;
    72→  for (const r of ranges) {
    73→    if (r.start >= lastEnd) {
    74→      out.push(r);
    75→      lastEnd = r.end;
    76→    }
    77→  }
    78→  return out;
    79→}

    81→/**
    82→ * Resolve a citation marker inner-text (e.g. "1", "2,3", "PMID:12345") to one
    83→ * or more reference records. Numeric markers map to the 1-indexed references
    84→ * array; SOURCE:ID markers match by type+externalId.
    85→ */
    86→function resolveCitation(
    87→  inner: string,
    88→  references: CitationRef[]
    89→): CitationRef[] {
    90→  const trimmed = inner.trim();
    91→  // SOURCE:ID form
    92→  const srcMatch = trimmed.match(/^([A-Z]{2,12}):\s?(.+)$/);
    93→  if (srcMatch) {
    94→    const rawSource = srcMatch[1].toLowerCase();
    95→    // Normalize aliases
    96→    const source =
    97→      rawSource === "pmid" ? "pubmed" :
    98→      rawSource === "pdb" ? "rcsb" :
    99→      rawSource;
   100→    const id = srcMatch[2].trim();
   101→    const found = references.find(
   102→      (r) => {
   103→        const rType =
   104→          r.type?.toLowerCase() === "pmid" ? "pubmed" :
   105→          r.type?.toLowerCase() === "pdb" ? "rcsb" :
   106→          r.type?.toLowerCase();
   107→        return rType === source && (
   108→          r.externalId?.toLowerCase() === id.toLowerCase() ||
   109→          r.externalId?.toLowerCase().includes(id.toLowerCase()) ||
   110→          id.toLowerCase().includes(r.externalId?.toLowerCase() || "___")
   111→        );
   112→      }
   113→    );
   114→    return found ? [found] : [];
   115→  }
   116→  // Numeric / range form, e.g. "1", "2,3", "1-3", "2–4"
   117→  const nums: number[] = [];
   118→  const parts = trimmed.split(/[,;]\s*/);
   119→  for (const p of parts) {
   120→    const rangeMatch = p.match(/^(\d+)\s*[-–]\s*(\d+)$/);
   121→    if (rangeMatch) {
   122→      const a = parseInt(rangeMatch[1], 10);
   123→      const b = parseInt(rangeMatch[2], 10);
   124→      for (let n = a; n <= b; n++) nums.push(n);
   125→    } else {
   126→      const n = parseInt(p, 10);
   127→      if (!isNaN(n)) nums.push(n);
   128→    }
   129→  }
   130→  return nums
   131→    .map((n) => references[n - 1])
   132→    .filter(Boolean) as CitationRef[];
   133→}

   135→/**
   136→ * Parse the AI-generated "### Citations" block OR the article "## References"
   137→ * section into structured references. Both use the same "[n] ..." format.
   138→ * Returns a sparse array indexed by the citation number (so [1] = index 0,
   139→ * [3] = index 2, with undefined for gaps like [9] if not cited).
   140→ */
   141→export function parseCitationsBlock(text: string): CitationRef[] {
   142→  const lines = text.split("\n").filter((l) => l.trim());
   143→  const refMap: Map<number, CitationRef> = new Map();
   144→  let maxNum = 0;
   145→  for (const line of lines) {
   146→    const m = line.match(/^\s*\[(\d+)\]\s*(.+)$/);
   147→    if (!m) continue;
   148→    const num = parseInt(m[1], 10);
   149→    if (num > maxNum) maxNum = num;
   150→    const body = m[2].trim();
   151→
   152→    // Extract URL (— https://... or just https://...)
   153→    const urlMatch = body.match(/https?:\/\/[^\s]+/);
   154→    let url = urlMatch?.[1]?.replace(/[—–-]\s*$/, "").trim();
   155→
   156→    // Extract DOI (doi:...)
   157→    const doiMatch = body.match(/doi:(10\.\S+)/i);
   158→    const doi = doiMatch?.[1]?.replace(/[.,;]\s*$/, "");
   159→
   160→    // Extract year (YYYY)
   161→    const yearMatch = body.match(/\((\d{4}[a-z]?)\)/);
   162→    const year = yearMatch?.[1];
   163→
   164→    // Extract PMID (pubmed:NNNNN or PMID:NNNNN)
   165→    const pmidMatch = body.match(/(?:pubmed|PMID)[:\s]+(\d+)/i);
   166→    const pmid = pmidMatch?.[1];
   167→
   168→    // Extract source:ID (PDB:XXXX, PMID:NNNNN, etc.)
   169→    const sourceMatch =
   170→      body.match(/\[([A-Z]{2,12}):\s?([^\]]+)\]/) ||
   171→      body.match(/\b([A-Z]{2,12}):\s?([A-Za-z0-9_\-\.]+)/);
   172→    const rawType = sourceMatch?.[1]?.toLowerCase();
   173→    const type =
   174→      rawType === "pmid" ? "pubmed" :
   175→      rawType === "pdb" ? "rcsb" :
   176→      rawType;
   177→    const externalId = sourceMatch?.[2]?.trim() || pmid;
   178→
   179→    // Build URL from PMID or source:ID if not explicitly provided
   180→    if (!url && pmid) {
   181→      url = `https://pubmed.ncbi.nlm.nih.gov/${pmid}/`;
   182→    } else if (!url && type && externalId) {
   183→      const id = externalId.trim();
   184→      if (type === "pubmed" || type === "pmid")
   185→        url = `https://pubmed.ncbi.nlm.nih.gov/${id}/`;
   186→      else if (type === "pmc") url = `https://www.ncbi.nlm.nih.gov/pmc/articles/${id}/`;
   187→      else if (type === "uniprot") url = `https://www.uniprot.org/uniprotkb/${id}`;
   188→      else if (type === "rcsb" || type === "pdb") url = `https://www.rcsb.org/structure/${id}`;
   189→      else if (type === "ncbi" || type === "gene") url = `https://www.ncbi.nlm.nih.gov/gene/${id}`;
   190→      else if (type === "doi") url = `https://doi.org/${id}`;
   191→    }
   192→
   193→    // Parse "Authors (Year) Journal. Title. — URL" format
   194→    // The body format is: "Authors (Year) Journal. Title. — URL"
   195→    // or: "Authors (Year), Journal. Title. [SOURCE:ID] — URL"
   196→    let authors: string | undefined;
   197→    let journal: string | undefined;
   198→    let title: string = body;
   199→
   200→    // Try to extract authors (everything before the year)
   201→    if (yearMatch && yearMatch.index !== undefined) {
   202→      authors = body.slice(0, yearMatch.index).trim().replace(/[,\s]+$/, "");
   203→      const afterYear = body.slice(yearMatch.index + yearMatch[0].length).trim();
   204→
   205→      // Try to extract journal (text between year and the title, ending with .)
   206→      // Format: "(Year) Journal. Title." or "(Year), Journal. Title."
   207→      const journalMatch = afterYear.match(/^,?\s*([^.,]+)\.\s+(.+)$/);
   208→      if (journalMatch) {
   209→        journal = journalMatch[1].trim();
   210→        title = journalMatch[2].trim();
   211→      } else {
   212→        title = afterYear;
   213→      }
   214→    }
   215→
   216→    // Clean up title: remove source:ID, URL, DOI
   217→    if (sourceMatch) {
   218→      title = title.replace(/\[?[A-Z]{2,12}:\s?[^\]\s]+]?/g, "");
   219→    }
   220→    title = title
   221→      .replace(/https?:\/\/[^\s]+/g, "")
   222→      .replace(/doi:\S+/gi, "")
   223→      .replace(/[—–-]\s*$/, "")
   224→      .replace(/^\s*[—–-]\s*/, "")
   225→      .replace(/\.$/, "")
   226→      .trim();
   227→
   228→    // If all we have is a source:ID, construct a title from it
   229→    const fallbackTitle = type && externalId ? `${type.toUpperCase()}:${externalId}` : body.slice(0, 200);
   230→
   231→    refMap.set(num, {
   232→      type: type || "manual",
   233→      externalId,
   234→      title: title.slice(0, 200) || fallbackTitle,
   235→      year,
   236→      url,
   237→      doi,
   238→      authors,
   239→      journal,
   240→    });
   241→  }
   242→  // Return a sparse array: index 0 = ref [1], index 8 = ref [9] (or undefined if not cited)
   243→  const result: CitationRef[] = [];
   244→  for (let k = 1; k <= maxNum; k++) {
   245→    result.push(refMap.get(k) || null as any);
   246→  }
   247→  return result;
   248→}

   250→/**
   251→ * Render content segments as properly structured HTML:
   252→ * - Lines starting with ## or ### become styled headings (h2/h3)
   253→ * - Other content becomes paragraphs with inline citations/highlights
   254→ */
   255→function renderContentBlocks(
   256→  segments: Segment[],
   257→  ctx: {
   258→    allRefs: CitationRef[];
   259→    onCitationClick?: (ref: CitationRef, index: number) => void;
   260→    onAnnotationClick?: (a: Annotation) => void;
   261→  }
   262→): React.ReactNode {
   263→  const { allRefs, onCitationClick, onAnnotationClick } = ctx;
   264→
   265→  // Group segments into blocks: heading blocks and paragraph blocks
   266→  const blocks: { type: "heading" | "paragraph"; level: number; segments: Segment[] }[] = [];
   267→  let currentBlock: { type: "heading" | "paragraph"; level: number; segments: Segment[] } | null = null;
   268→
   269→  for (const seg of segments) {
   270→    const text = seg.text;
   271→    const headingMatch = text.match(/^(#{2,4})\s+(.+)$/m);
   272→
   273→    if (seg.type === "text" && headingMatch) {
   274→      if (currentBlock) {
   275→        blocks.push(currentBlock);
   276→        currentBlock = null;
   277→      }
   278→
   279→      const headingLine = headingMatch[0];
   280→      const headingLevel = headingMatch[1].length;
   281→      const headingText = headingMatch[2];
   282→      const beforeHeading = text.slice(0, headingMatch.index).trim();
   283→      const afterHeading = text.slice(headingMatch.index! + headingLine.length).trim();
   284→
   285→      if (beforeHeading) {
   286→        blocks.push({ type: "paragraph", level: 0, segments: [{ type: "text", text: beforeHeading }] });
   287→      }
   288→      blocks.push({ type: "heading", level: headingLevel, segments: [{ type: "text", text: headingText }] });
   289→
   290→      if (afterHeading) {
   291→        currentBlock = { type: "paragraph", level: 0, segments: [{ type: "text", text: afterHeading }] };
   292→      }
   293→    } else {
   294→      if (!currentBlock || currentBlock.type === "heading") {
   295→        if (currentBlock) blocks.push(currentBlock);
   296→        currentBlock = { type: "paragraph", level: 0, segments: [] };
   297→      }
   298→      currentBlock.segments.push(seg);
   299→    }
   300→  }
   301→  if (currentBlock) blocks.push(currentBlock);
   302→
   303→  if (blocks.length === 0) {
   304→    return (
   305→      <p className="whitespace-pre-wrap break-words m-0">
   306→        {segments.map((s, idx) => renderSegment(s, idx, allRefs, onCitationClick, onAnnotationClick))}
   307→      </p>
   308→    );
   309→  }
   310→
   311→  return blocks.map((block, bi) => {
   312→    if (block.type === "heading") {
   313→      const level = block.level;
   314→      const headingText = block.segments[0]?.text || "";
   315→      const headingClass = level === 2
   316→        ? "text-base font-serif-text font-semibold text-foreground mt-5 mb-2 pb-1 border-b border-border/40"
   317→        : level === 3
   318→        ? "text-sm font-semibold text-foreground/90 mt-4 mb-1.5"
   319→        : "text-xs font-semibold text-muted-foreground mt-3 mb-1 uppercase tracking-wide";
   320→      return React.createElement(
   321→        level === 2 ? "h2" : level === 3 ? "h3" : "h4",
   322→        { key: `heading-${bi}`, className: headingClass },
   323→        headingText
   324→      );
   325→    }
   326→    return (
   327→      <p key={`para-${bi}`} className="whitespace-pre-wrap break-words m-0 mb-3 leading-relaxed">
   328→        {block.segments.map((s, idx) => renderSegment(s, bi * 1000 + idx, allRefs, onCitationClick, onAnnotationClick))}
   329→      </p>
   330→    );
   331→  });
   332→}

   334→/** Render a single segment (cite, highlight, or text) with full HoverCard support */
   335→/**
   336→ * Parse inline markdown in a text segment and return React nodes.
   337→ *
   338→ * Supports:
   339→ *   **bold**    → <strong>
   340→ *   *italic*    → <em>
   341→ *   `code`      → <code>
   342→ *
   343→ * Bold is matched before italic (** is greedier than *). Code is matched
   344→ * first so that `**not bold**` inside backticks stays literal.
   345→ *
   346→ * This is a lightweight inline parser — it does NOT handle block-level
   347→ * markdown (headings, lists, links) because those are handled elsewhere
   348→ * (renderContentBlocks splits on ## headings). It only fixes the common
   349→ * case where the LLM emits **TMC1** or *C. elegans* in the body text and
   350→ * the raw asterisks would otherwise show up verbatim.
   351→ */
   352→function renderInlineMarkdown(text: string, baseKey: string): React.ReactNode[] {
   353→  const nodes: React.ReactNode[] = [];
   354→  // Regex matches: `code` OR **bold** OR *italic*
   355→  // Order matters: code first (so ** inside backticks is literal), then
   356→  // bold (**), then italic (*).
   357→  const re = /(`[^`]+`)|(\*\*[^*]+\*\*)|(\*[^*]+\*)/g;
   358→  let lastIndex = 0;
   359→  let matchIndex = 0;
   360→  let m: RegExpExecArray | null;
   361→  while ((m = re.exec(text)) !== null) {
   362→    // Push preceding plain text
   363→    if (m.index > lastIndex) {
   364→      nodes.push(text.slice(lastIndex, m.index));
   365→    }
   366→    const token = m[0];
   367→    const key = `${baseKey}-${matchIndex++}`;
   368→    if (token.startsWith("`") && token.endsWith("`")) {
   369→      // inline code
   370→      nodes.push(
   371→        <code key={key} className="px-1 py-0.5 rounded bg-muted/60 text-[0.85em] font-mono">
   372→          {token.slice(1, -1)}
   373→        </code>,
   374→      );
   375→    } else if (token.startsWith("**") && token.endsWith("**")) {
   376→      // bold
   377→      nodes.push(
   378→        <strong key={key} className="font-semibold text-foreground">
   379→          {token.slice(2, -2)}
   380→        </strong>,
   381→      );
   382→    } else if (token.startsWith("*") && token.endsWith("*")) {
   383→      // italic
   384→      nodes.push(
   385→        <em key={key} className="italic">
   386→          {token.slice(1, -1)}
   387→        </em>,
   388→      );
   389→    }
   390→    lastIndex = m.index + token.length;
   391→  }
   392→  // Push trailing plain text
   393→  if (lastIndex < text.length) {
   394→    nodes.push(text.slice(lastIndex));
   395→  }
   396→  return nodes.length > 0 ? nodes : [text];
   397→}

   399→function renderSegment(
   400→  s: Segment,
   401→  idx: number,
   402→  allRefs: CitationRef[],
   403→  onCitationClick?: (ref: CitationRef, index: number) => void,
   404→  onAnnotationClick?: (a: Annotation) => void
   405→): React.ReactNode {
   406→  if (s.type === "cite") {
   407→    const inner = s.text.replace(/^\[|\]$/g, "");
   408→    const refs = resolveCitation(inner, allRefs);
   409→    const firstRef = refs[0];
   410→    // Layer 3 rendering guard: detect unresolved / suspect citations.
   411→    // - No refs resolved at all → the [n] points to nothing (hallucinated or
   412→    //   out-of-range). Render the marker in red with a "?" suffix.
   413→    // - A resolved ref carries an auditStatus of "suspect"/"unsupported"/
   414→    //   "missing" → render with a warning icon + colored ring.
   415→    const isUnresolved = refs.length === 0;
   416→    const auditFlag = refs.find(
   417→      (r) => r.auditStatus && r.auditStatus !== "ok"
   418→    )?.auditStatus;
   419→    const auditReason = refs.find(
   420→      (r) => r.auditStatus && r.auditStatus !== "ok"
   421→    )?.auditReason;
   422→    const markerClass = isUnresolved
   423→      ? "cite-marker cite-marker-unresolved cursor-pointer"
   424→      : auditFlag
   425→      ? `cite-marker cite-marker-${auditFlag} cursor-pointer`
   426→      : "cite-marker cursor-pointer";
   427→    return (
   428→      <HoverCard key={idx} openDelay={120} closeDelay={120}>
   429→        <HoverCardTrigger asChild>
   430→          <span
   431→            className={markerClass}
   432→            tabIndex={0}
   433→            onClick={(e) => {
   434→              e.stopPropagation();
   435→              if (firstRef && onCitationClick) {
   436→                const refIndex = allRefs.findIndex(
   437→                  (r) =>
   438→                    (r.externalId === firstRef.externalId &&
   439→                      r.type === firstRef.type) ||
   440→                    r.title === firstRef.title
   441→                );
   442→                onCitationClick(firstRef, refIndex);
   443→              }
   444→            }}
   445→          >
   446→            {isUnresolved ? `${inner}?` : inner}
   447→            {auditFlag && (
   448→              <span className="cite-audit-icon" aria-hidden="true">
   449→                ⚠
   450→              </span>
   451→            )}
   452→          </span>
   453→        </HoverCardTrigger>
   454→        <HoverCardContent
   455→          side="top"
   456→          align="center"
   457→          className="w-72 p-3 text-xs shadow-lg z-50 rounded-lg border border-border/60 bg-gradient-to-br from-card to-muted/10"
   458→        >
   459→          {refs.length > 0 ? (
   460→            <div className="space-y-1.5">
   461→              {refs.map((r, ri) => (
   462→                <div key={ri} className="space-y-0.5">
   463→                  <p className="font-semibold leading-snug font-sans text-[11px]">
   464→                    {r.title}
   465→                  </p>
   466→                  <p className="text-[10px] text-muted-foreground leading-snug">
   467→                    {r.authors && <span>{r.authors}</span>}
   468→                    {r.authors && (r.year || r.journal) && (
   469→                      <span> · </span>
   470→                    )}
   471→                    {r.year && <span>{r.year}</span>}
   472→                    {r.journal && (
   473→                      <span className="italic"> {r.journal}</span>
   474→                    )}
   475→                  </p>
   476→                  {r.abstract && (
   477→                    <p className="text-[9px] text-muted-foreground/80 leading-relaxed line-clamp-3 italic">
   478→                      {r.abstract.slice(0, 200)}{r.abstract.length > 200 ? "…" : ""}
   479→                    </p>
   480→                  )}
   481→                  <div className="flex flex-wrap items-center gap-1">
   482→                    {r.type && r.externalId && (
   483→                      <span className="badge-slate px-1 py-0.5 rounded text-[8px] font-semibold uppercase">
   484→                        {r.type}:{r.externalId}
   485→                      </span>
   486→                    )}
   487→                    {r.doi && (
   488→                      <span className="text-[9px] font-mono text-muted-foreground">
   489→                        DOI:{r.doi}
   490→                      </span>
   491→                    )}
   492→                    {r.url && (
   493→                      <a
   494→                        href={r.url}
   495→                        target="_blank"
   496→                        rel="noreferrer"
   497→                        onClick={(e) => e.stopPropagation()}
   498→                        className="text-[9px] text-primary hover:underline inline-flex items-center gap-0.5"
   499→                      >
   500→                        <ExternalLink className="h-2.5 w-2.5" /> open
   501→                      </a>
   502→                    )}
   503→                  </div>
   504→                </div>
   505→              ))}
   506→            </div>
   507→          ) : (
   508→            <div className="space-y-1">
   509→              <p className="font-semibold text-[11px] text-red-600 dark:text-red-400">
   510→                Unresolved citation [{inner}]
   511→              </p>
   512→              <p className="text-[10px] text-muted-foreground">
   513→                No matching reference record found. This citation may be
   514→                hallucinated or out of range — it will be flagged by the
   515→                adversarial citation audit.
   516→              </p>
   517→            </div>
   518→          )}
   519→          {/* When refs resolved but flagged by the audit (suspect/unsupported),
   520→              append the audit reason to the hover card so the user sees WHY. */}
   521→          {auditFlag && auditReason && refs.length > 0 && (
   522→            <div className="mt-1.5 pt-1.5 border-t border-amber-300/40 text-[9px] text-amber-700 dark:text-amber-400 leading-snug">
   523→              <span className="font-semibold">Audit:</span> {auditReason}
   524→            </div>
   525→          )}
   526→        </HoverCardContent>
   527→      </HoverCard>
   528→    );
   529→  }
   530→  if (s.type === "highlight" && s.annotation) {
   531→    return (
   532→      <span
   533→        key={idx}
   534→        className={`ann-highlight ${
   535→          SEVERITY_TO_CLASS[s.annotation.severity] || ""
   536→        }`}
   537→        title={`${s.annotation.type}: ${s.annotation.comment}`}
   538→        onClick={(e) => {
   539→          e.stopPropagation();
   540→          onAnnotationClick?.(s.annotation!);
   541→        }}
   542→      >
   543→        {s.text}
   544→      </span>
   545→    );
   546→  }
   547→  // Text segment: parse inline markdown (**bold**, *italic*, `code`) so
   548→  // that the LLM's formatting markers don't show up as literal asterisks.
   549→  return <span key={idx}>{renderInlineMarkdown(s.text, `seg-${idx}`)}</span>;
   550→}
   551→
   552→export function MarkdownCitations({
   553→  content,
   554→  annotations = [],
   555→  references = [],
   556→  onAnnotationClick,
   557→  onCitationClick,
   558→  className = "",
   559→  suppressRefList = false,
   560→}: {
   561→  content: string;
   562→  annotations?: Annotation[];
   563→  references?: CitationRef[];
   564→  onAnnotationClick?: (a: Annotation) => void;
   565→  onCitationClick?: (ref: CitationRef, index: number) => void;
   566→  className?: string;
   567→  /** When true, suppress the component-generated reference list at the bottom.
   568→   * Used by VirtualizedArticle for body sections — the article's ## References
   569→   * section (rendered as its own section) already contains the full list. */
   570→  suppressRefList?: boolean;
   571→}) {
   572→  const { bodySegments, citationsBlock, contentRefText, citedRefs, allRefs, hasContentRefs } = React.useMemo(() => {
   573→    const highlights = buildHighlightRanges(content, annotations);
   574→    const citeRe = new RegExp(CITE_RE_SOURCE, "g");
   575→    const segments: Segment[] = [];
   576→    const citedList: CitationRef[] = [];
   577→    const citedKeys = new Set<string>();
   578→
   579→    // Pre-parse the AI "### Citations" block (if any) to build a fallback
   580→    // reference list for hover tooltips + the bottom reference list.
   581→    const citeHeaderIdx = content.indexOf("### Citations");
   582→    // Also check for "## References" header (articles have this).
   583→    // Also check for "## 参考文献" (Chinese reference section header) so that
   584→    // Chinese-translated articles (contentZh) can resolve [n] citations.
   585→    const refHeaderIdx =
   586→      content.indexOf("## References") >= 0
   587→        ? content.indexOf("## References")
   588→        : content.indexOf("## 参考文献");
   589→    // Also check for bare "REFERENCES" header
   590→    const bareRefIdx = content.indexOf("\nREFERENCES\n");
   591→
   592→    // The earliest reference/citations section header
   593→    const refSectionIdx = Math.min(
   594→      ...[citeHeaderIdx, refHeaderIdx, bareRefIdx].filter((i) => i >= 0),
   595→      content.length
   596→    );
   597→
   598→    // Parse BOTH "### Citations" AND "## References" blocks for hover tooltips.
   599→    // The "### Citations" block is paragraph-level; "## References" is article-level.
   600→    // We parse whichever one exists (or both) to build the reference list for
   601→    // hover tooltips. This ensures citations like [1], [2] etc. resolve to the
   602→    // correct reference even when the content has a "## References" section
   603→    // instead of a "### Citations" block.
   604→    const aiCitationsText =
   605→      citeHeaderIdx >= 0 ? content.slice(citeHeaderIdx) : "";
   606→    const parsedAiRefs = aiCitationsText ? parseCitationsBlock(aiCitationsText) : [];
   607→
   608→    // Also parse the "## References" or "REFERENCES" section if it exists.
   609→    // This is the article-level reference list that the compose step appends.
   610→    let parsedArticleRefs: CitationRef[] = [];
   611→    if (refHeaderIdx >= 0) {
   612→      const refText = content.slice(refHeaderIdx);
   613→      parsedArticleRefs = parseCitationsBlock(refText);
   614→    } else if (bareRefIdx >= 0) {
   615→      const refText = content.slice(bareRefIdx + 1); // +1 to skip the \n
   616→      parsedArticleRefs = parseCitationsBlock(refText);
   617→    }
   618→
   619→    // Merge: article refs (from ## References) take priority, then AI-parsed
   620→    // refs, then saved DB refs (which are indexed differently and should NOT
   621→    // override article-level references).
   622→    // CRITICAL: The `references` prop from page.tsx is a flat array of ALL
   623→    // project references (including web search results with paragraphId=null).
   624→    // These are indexed by insertion order, NOT by citation number. Using them
   625→    // as references[0] = [1] causes [1] to resolve to a random web search
   626→    // result instead of the article's actual first reference.
   627→    // Fix: Only use `references` prop when there are NO article-level refs
   628→    // AND NO AI-parsed refs (i.e., for standalone paragraph views without
   629→    // a composed article).
   630→    // Merge article-level refs (## References) and AI-parsed refs (### Citations).
   631→    // Gaps (sparse entries) are filled with a `missing` sentinel so the
   632→    // rendering guard (Layer 3) can flag the corresponding [n] marker in red
   633→    // instead of fabricating a misleading "Reference N" tooltip.
   634→    const merged: CitationRef[] = [];
   635→    const maxLen = Math.max(parsedArticleRefs.length, parsedAiRefs.length);
   636→    for (let k = 0; k < maxLen; k++) {
   637→      const articleRef = parsedArticleRefs[k];
   638→      const ai = parsedAiRefs[k];
   639→      merged.push(
   640→        articleRef || ai || {
   641→          type: "missing",
   642→          title: `Reference ${k + 1} (missing)`,
   643→          auditStatus: "missing",
   644→          auditReason: `No entry [${k + 1}] found in the References section.`,
   645→        }
   646→      );
   647→    }
   648→    // If no article/AI refs were found, fall back to the `references` prop
   649→    // (for standalone paragraph rendering without a composed article).
   650→    if (merged.length === 0 && references.length > 0) {
   651→      merged.push(...references);
   652→    }
   653→
   654→    const pushCited = (ref: CitationRef) => {
   655→      const key = ref.id || `${ref.type}:${ref.externalId}` || ref.title;
   656→      if (!citedKeys.has(key)) {
   657→        citedKeys.add(key);
   658→        citedList.push(ref);
   659→      }
   660→    };
   661→
   662→    let i = 0;
   663→    while (i < content.length) {
   664→      const hl = highlights.find((h) => h.start === i);
   665→      if (hl) {
   666→        segments.push({
   667→          type: "highlight",
   668→          text: content.slice(hl.start, hl.end),
   669→          annotation: hl.annotation,
   670→        });
   671→        i = hl.end;
   672→        continue;
   673→      }
   674→      citeRe.lastIndex = i;
   675→      const m = citeRe.exec(content);
   676→      if (m && m.index === i) {
   677→        const inner = m[1];
   678→        const refs = resolveCitation(inner, merged);
   679→        refs.forEach(pushCited);
   680→        segments.push({ type: "cite", text: m[0], citeKey: inner });
   681→        i += m[0].length;
   682→        continue;
   683→      }
   684→      let nextStop = content.length;
   685→      for (const h of highlights) {
   686→        if (h.start > i) {
   687→          nextStop = Math.min(nextStop, h.start);
   688→          break;
   689→        }
   690→      }
   691→      citeRe.lastIndex = i + 1;
   692→      const m2 = citeRe.exec(content);
   693→      if (m2 && m2.index > i) {
   694→        nextStop = Math.min(nextStop, m2.index);
   695→      }
   696→      segments.push({ type: "text", text: content.slice(i, nextStop) });
   697→      i = nextStop;
   698→    }
   699→
   700→    // Split body from reference/citations section at the earliest header found.
   701→    // The content may have "### Citations" (paragraph) or "## References" (article)
   702→    // or bare "REFERENCES" (AI-generated). We split at the earliest one.
   703→    if (refSectionIdx >= content.length) {
   704→      // No reference section header found — entire content is body
   705→      return { bodySegments: segments, citationsBlock: null, citedRefs: citedList, allRefs: merged, hasContentRefs: false };
   706→    }
   707→    let acc = 0;
   708→    const body: Segment[] = [];
   709→    const rest: Segment[] = [];
   710→    for (const s of segments) {
   711→      const segStart = acc;
   712→      const segEnd = acc + s.text.length;
   713→      if (segEnd <= refSectionIdx) {
   714→        body.push(s);
   715→      } else if (segStart >= refSectionIdx) {
   716→        rest.push(s);
   717→      } else {
   718→        const splitAt = refSectionIdx - segStart;
   719→        body.push({ ...s, text: s.text.slice(0, splitAt) });
   720→        rest.push({ ...s, text: s.text.slice(splitAt) });
   721→      }
   722→      acc = segEnd;
   723→    }
   724→    const citText = rest.map((s) => s.text).join("");
   725→    // Only treat as citationsBlock (for fallback rendering) if it's a "### Citations" block.
   726→    // For "## References" or "REFERENCES", we render the text as-is via contentRefText.
   727→    const isCitationsBlock = citeHeaderIdx >= 0 && citeHeaderIdx === refSectionIdx;
   728→    return {
   729→      bodySegments: body,
   730→      citationsBlock: isCitationsBlock ? citText : null,
   731→      contentRefText: !isCitationsBlock ? citText : null,
   732→      citedRefs: citedList,
   733→      allRefs: merged,
   734→      hasContentRefs: !isCitationsBlock,
   735→    };
   736→  }, [content, annotations, references]);
   737→
   738→  return (
   739→    <div className={`prose-academic ${className}`}>
   740→      {/* Render content blocks: headings as styled h2/h3, body as paragraph */}
   741→      {renderContentBlocks(bodySegments, { allRefs, onCitationClick, onAnnotationClick })}
   742→
   743→      {/* Fallback: raw AI-generated citation block (only if no structured refs) */}
   744→      {citationsBlock && citedRefs.length === 0 && (
   745→        <div className="mt-3 pt-3 border-t border-dashed border-border/70 text-xs text-muted-foreground font-mono whitespace-pre-wrap leading-relaxed">
   746→          {citationsBlock}
   747→        </div>
   748→      )}
   749→
   750→      {/* Article's own reference section (from "## References" or "REFERENCES" in content).
   751→          Rendered as literal text — no duplicate component-generated list is added. */}
   752→      {contentRefText && (
   753→        <div className="mt-4 pt-3 border-t border-border/70">
   754→          <div className="text-[11px] leading-snug font-sans text-foreground/85 whitespace-pre-wrap">
   755→            {contentRefText}
   756→          </div>
   757→        </div>
   758→      )}
   759→
   760→      {/* Component-generated reference list — only shown if the content does NOT
   761→          already have its own reference section AND suppressRefList is false. */}
   762→      {!hasContentRefs && !suppressRefList && allRefs.length > 0 && (
   763→        <div className="mt-4 pt-3 border-t border-border/70">
   764→          <p className="divider-academic mb-2">
   765→            <span>References</span>
   766→          </p>
   767→          {/* v97-1: Enhanced reference list with rounded-lg + shadow-sm */}
   768→          <ol className="list-none rounded-lg overflow-hidden border border-border/40 shadow-sm">
   769→            {allRefs.map((r, i) => (
   770→              <li
   771→                id={`ref-${i + 1}`}
   772→                key={r.id || i}
   773→                className={`text-[11px] leading-snug flex gap-1.5 font-sans text-foreground/85 px-2.5 py-1.5 transition-colors hover:bg-accent/30 scroll-mt-16 ${
   774→                  r.auditStatus === "missing"
   775→                    ? "bg-red-50/60 dark:bg-red-950/20"
   776→                    : r.auditStatus === "unsupported" || r.auditStatus === "suspect"
   777→                    ? "bg-amber-50/50 dark:bg-amber-950/15"
   778→                    : i % 2 === 0
   779→                    ? "bg-transparent"
   780→                    : "bg-muted/25"
   781→                }`}
   782→              >
   783→                <span className="font-mono text-primary font-semibold shrink-0">
   784→                  [{i + 1}]
   785→                </span>
   786→                <span className="flex-1">
   787→                  {r.authors && <span>{r.authors} </span>}
   788→                  {r.year && (
   789→                    <span className="text-muted-foreground">({r.year}) </span>
   790→                  )}
   791→                  <span className="font-medium">{r.title}.</span>
   792→                  {r.journal && (
   793→                    <span className="italic text-muted-foreground">
   794→                      {" "}
   795→                      {r.journal}.
   796→                    </span>
   797→                  )}
   798→                  {r.type && r.externalId && (
   799→                    <span className="ml-1 badge-slate px-1 py-0.5 rounded text-[8px] font-semibold uppercase align-middle">
   800→                      {r.type}:{r.externalId}
   801→                    </span>
   802→                  )}
   803→                  {r.doi && (
   804→                    <span className="ml-1 text-[9px] font-mono text-muted-foreground">
   805→                      doi:{r.doi}
   806→                    </span>
   807→                  )}
   808→                  {r.url && (
   809→                    <a
   810→                      href={r.url}
   811→                      target="_blank"
   812→                      rel="noreferrer"
   813→                      className="ml-1 text-primary hover:underline inline-flex items-center gap-0.5 text-[9px]"
   814→                    >
   815→                      <ExternalLink className="h-2.5 w-2.5" />
   816→                    </a>
   817→                  )}
   818→                </span>
   819→              </li>
   820→            ))}
   821→          </ol>
   822→        </div>
   823→      )}
   824→    </div>
   825→  );
   826→}
